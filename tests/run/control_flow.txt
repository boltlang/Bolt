================================================================================
Breaking out of an outer loop
:test-prelude
================================================================================
let mut steps = 0;
let mut i = 0;
'outer: while i < 3 {
  i += 1;
  let mut j = 0;
  while j < 3 {
    j += 1;
    steps += 1;
    if i * j == 4 { break 'outer; }
  }
}
print(show(steps));
--------------------------------------------------------------------------------
5
