================================================================================
Indexing arrays
:test-prelude
================================================================================
let xs = [10, 20, 30];
xs[1] + xs[2];
--------------------------------------------------------------------------------
50

================================================================================
Changing elements
:test-prelude
================================================================================
let xs = mut [1, 2, 3];
xs[0] = 7;
xs[1] += 5;
xs;
--------------------------------------------------------------------------------
[7, 7, 3]

================================================================================
Elements of every size
:test-prelude
================================================================================
let bytes = mut [1u8, 255u8, 0u8];
bytes[2] = 128u8;
let small = [-5i8, 7i8];
let wide = [1i64 << 40, -1i64];
let flags = mut [True, False];
flags[1] = True;
(bytes, small[0], wide[0] + wide[1], flags, [1.5, 2.5][1], array_len([(), ()]));
--------------------------------------------------------------------------------
([1, 255, 128], -5, 1099511627775, [True, True], 2.5, 2)

================================================================================
Arrays of arrays, structs and tuples
:test-prelude
================================================================================
struct P { x: i32 }
let nested = [[1, 2], [3]];
let ps = mut [mut P { x: 1 }, mut P { x: 2 }];
ps[1].x = 5;
let ts = mut [(1, 2)];
ts[0].1 = 5;
(nested[0][1] + nested[1][0], ps[0].x + ps[1].x, ts);
--------------------------------------------------------------------------------
(5, 6, [(1, 5)])

================================================================================
Generic functions over arrays
:test-prelude
================================================================================
fn first[T](xs: [T]) -> T = xs[0];
fn sum(xs: [i32]) -> i32 {
  let mut total = 0;
  let mut i = 0;
  while i < array_len(xs) {
    total += xs[i];
    i += 1;
  }
  total
}
(first([7, 8]), first([True]), sum([1, 2, 3, 4]));
--------------------------------------------------------------------------------
(7, True, 10)

================================================================================
Copying ranges, within an array too
:prelude
================================================================================
let a = mut [1, 2, 3, 4, 5];
let b = mut [0, 0, 0];
array_copy(b, 1, a, 3, 2);
array_copy(a, 1, a, 0, 3);
(b, a);
--------------------------------------------------------------------------------
([0, 4, 5], [1, 1, 2, 3, 5])

================================================================================
Copies that can and can't be changed
:prelude
================================================================================
let xs = mut [1, 2];
let frozen = to_imm(xs);
let copy = to_mut(frozen);

copy[1] = 8;
