================================================================================
Generic functions over arrays
:test-prelude
================================================================================
fn first[T](xs: [T]) -> T = xs[0];
fn sum(xs: [i32]) -> i32 {
  let mut total = 0;
  let mut i = 0;
  while i < array_len(xs) {
    total += xs[i];
    i += 1;
  }
  total
}
print(show((first([7, 8]), first([True]), sum([1, 2, 3, 4]))));
--------------------------------------------------------------------------------
(7, True, 10)
