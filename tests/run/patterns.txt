fn classify(n: i32) -> i32 = match n {
  0 => 0,
  1 | 2 | 3 => 1,
  4..=9 => 2,
  x if x < 0 => -1,
  _ => 3,
};
classify(0) * 10000 + classify(2) * 1000 + classify(7) * 100 + classify(100) * 10 + classify(-5);

fn half(n: i32) -> i32 {
  let 0..=9 = n else { return -1; };
  n / 2
}
