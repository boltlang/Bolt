================================================================================
A generic function chooses the impl when it runs
:test-prelude
================================================================================
class Describe[T] { fn describe(x: T) -> String; }
impl Describe[i32] { fn describe(x) = "a number"; }
impl Describe[Bool] { fn describe(x) = "a boolean"; }
fn shout(x) = describe(x);
fn twice[T](x: T) -> (String, String) where Describe[T] = (describe(x), describe(x));
print(show((shout(1), shout(True), twice(False))));
--------------------------------------------------------------------------------
(a number, a boolean, (a boolean, a boolean))

================================================================================
A generic impl calls the impl of its parameter
:test-prelude
================================================================================
enum Option[T] { None, Some(T) }
import Option.*;
class Describe[T] { fn describe(x: T) -> String; }
impl Describe[i32] { fn describe(x) = "a number"; }
impl[T] Describe[Option[T]] where Describe[T] {
  fn describe(x) = match x { Some(y) => describe(y), None => "nothing" };
}
let nothing: Option[i32] = None;
print(show((describe(Some(1)), describe(nothing))));
--------------------------------------------------------------------------------
(a number, nothing)

================================================================================
An associated type
:test-prelude
================================================================================
class Get[C] {
  type Item;
  fn get(c: C) -> Item;
}
struct Box[T] { item: T }
impl[T] Get[Box[T]] {
  type Item = T;
  fn get(b) = b.item;
}
fn first[C](c: C) -> C.Item where Get[C] = get(c);
print(show((get(Box { item: True }), first(Box { item: 1 }))));
--------------------------------------------------------------------------------
(True, 1)

================================================================================
Looping over a type with its own iterator
:test-prelude
================================================================================
struct Countdown { from: i32 }
struct CountdownIter { left: i32 }
impl Iterator[mut CountdownIter] {
  type Item = i32;
  fn next(it) = if it.left > 0 {
    it.left -= 1;
    Some(it.left + 1)
  } else {
    None
  };
}
impl Iterable[Countdown] {
  type Iter = mut CountdownIter;
  fn iter(c) = mut CountdownIter { left: c.from };
}
let countdown = Countdown { from: 3 };
let mut digits = 0;
for n in countdown {
  digits = digits * 10 + n;
}
print(show(digits));
--------------------------------------------------------------------------------
321

================================================================================
An impl with a bound uses the impl of its type's argument
:prelude
================================================================================
class Name[T] { fn name(x: T) -> String; }
impl Name[i32] { fn name(x) = "i32"; }
impl Name[i64] { fn name(x) = "i64"; }
impl[T] Name[Option[T]] where Name[T] { fn name(x) = match x { Some(y) => "Some " <> name(y), None => "None" }; }
fn later(x) = || name(x);
print(later(Some(1))());
print(later(Some(2i64))());
--------------------------------------------------------------------------------
Some i32
Some i64
