================================================================================
Enums in parameters, results and matches
================================================================================
enum Color { Red, Green, Blue }
fn next(c: Color) -> Color = match c {
  Color.Red => Color.Green,
  Color.Green => Color.Blue,
  Color.Blue => Color.Red,
};
next(next(Color.Red));
--------------------------------------------------------------------------------
Blue

================================================================================
Variants imported into scope
:prelude
================================================================================
enum Color { Red, Green, Blue }
import Color.*;
fn code(c) = match c { Red => 1, Green => 2, Blue => 3 };
fn warm(c: Color) -> Bool = match c { Red | Green => True, _ => False };
print(if warm(Green) && !warm(Blue) { "warm" } else { "cold" });
code(Blue) * 10 + code(Red);
--------------------------------------------------------------------------------
warm
31

================================================================================
Equality of variants
================================================================================

Color.Red == Color.Red && Color.Red != Color.Blue;
--------------------------------------------------------------------------------

A variable that changes
================================================================================

================================================================================
Generic enums without fields
================================================================================
enum Tag[T] { A, B }
fn tag(x: i32) -> Tag[i32] = if x > 0 { Tag.B } else { Tag.A };
tag(5);
--------------------------------------------------------------------------------
B

================================================================================
A const variant in a pattern
================================================================================
enum Color { Red, Green, Blue }
const FAVORITE: Color = Color.Green;
fn is_favorite(c: Color) -> Bool = match c { FAVORITE => True, _ => False };
(FAVORITE == Color.Green) && is_favorite(Color.Green);
--------------------------------------------------------------------------------
True
