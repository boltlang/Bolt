================================================================================
Enums in parameters, results and matches
:test-prelude
================================================================================
enum Color { Red, Green, Blue }
impl Show[Color] {
  fn show(c) = match c { Color.Red => "Red", Color.Green => "Green", Color.Blue => "Blue" };
}
fn rotate(c: Color) -> Color = match c {
  Color.Red => Color.Green,
  Color.Green => Color.Blue,
  Color.Blue => Color.Red,
};
print(show(rotate(rotate(Color.Red))));
--------------------------------------------------------------------------------
Blue

================================================================================
Variants imported into scope
:test-prelude
================================================================================
enum Color { Red, Green, Blue }
import Color.*;
fn code(c) = match c { Red => 1, Green => 2, Blue => 3 };
fn warm(c: Color) -> Bool = match c { Red | Green => True, _ => False };
print(if warm(Green) && !warm(Blue) { "warm" } else { "cold" });
print(show(code(Blue) * 10 + code(Red)));
--------------------------------------------------------------------------------
warm
31

================================================================================
Equality of variants
:test-prelude
================================================================================
enum Color { Red, Green, Blue }
impl Eq[Color] {
  fn eq(a, b) = match (a, b) {
    (Color.Red, Color.Red) | (Color.Green, Color.Green) | (Color.Blue, Color.Blue) => True,
    _ => False,
  };
}
print(show(Color.Red == Color.Red && Color.Red != Color.Blue));
--------------------------------------------------------------------------------
True

================================================================================
A variable that changes
:test-prelude
================================================================================
enum Color { Red, Green, Blue }
impl Show[Color] {
  fn show(c) = match c { Color.Red => "Red", Color.Green => "Green", Color.Blue => "Blue" };
}
let mut c = Color.Red;
if let Color.Red = c { c = Color.Blue; }
print(show(c));
--------------------------------------------------------------------------------
Blue

================================================================================
Generic enums without fields
:test-prelude
================================================================================
enum Tag[T] { A, B }
impl[T] Show[Tag[T]] {
  fn show(t) = match t { Tag.A => "A", Tag.B => "B" };
}
fn tag(x: i32) -> Tag[i32] = if x > 0 { Tag.B } else { Tag.A };
print(show(tag(5)));
--------------------------------------------------------------------------------
B

================================================================================
A const variant in a pattern
:test-prelude
================================================================================
enum Color { Red, Green, Blue }
const FAVORITE: Color = Color.Green;
fn is_favorite(c: Color) -> Bool = match c { FAVORITE => True, _ => False };
print(show(is_favorite(Color.Green)));
--------------------------------------------------------------------------------
True

================================================================================
Enums with fields
:test-prelude
================================================================================
enum Shape { Circle(f64), Square(f64) }
fn area(s: Shape) -> f64 = match s {
  Shape.Circle(r) => 3.0 * r * r,
  Shape.Square(w) => w * w,
};
print(show(area(Shape.Square(2.0)) + area(Shape.Circle(1.0))));
--------------------------------------------------------------------------------
7.0

================================================================================
Variants with named fields and without fields
:test-prelude
================================================================================
enum Shape { Circle(f64), Rect { w: f64, h: f64 }, Empty }
fn area(s: Shape) -> f64 = match s {
  Shape.Circle(r) => 3.0 * r * r,
  Shape.Rect { w, h } => w * h,
  Shape.Empty => 0.0,
};
print(show((area(Shape.Circle(2.0)), area(Shape.Rect { w: 2.0, h: 3.0 }), area(Shape.Empty))));
--------------------------------------------------------------------------------
(12.0, 6.0, 0.0)

================================================================================
Nested patterns only look at the fields of the variant they match
:test-prelude
================================================================================
fn f(o: Option[Option[i32]]) -> i32 = match o {
  Some(Some(n)) => n,
  Some(None) => 0,
  None => -1,
};
print(show((f(Some(Some(3))), f(Some(None)), f(None), Some(Some(True)))));
--------------------------------------------------------------------------------
(3, 0, -1, Some(Some(True)))

================================================================================
A recursive enum
:test-prelude
================================================================================
enum List { Nil, Cons(i32, List) }
fn sum(l: List) -> i32 = match l {
  List.Nil => 0,
  List.Cons(x, rest) => x + sum(rest),
};
print(show(sum(List.Cons(1, List.Cons(2, List.Cons(3, List.Nil))))));
--------------------------------------------------------------------------------
6
