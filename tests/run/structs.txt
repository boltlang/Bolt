================================================================================
Structs are shared
:test-prelude
================================================================================
struct Point { x: i32, y: i32 }
fn shift(p: mut Point) { p.x += 1; }
let p = mut Point { x: 0, y: 0 };
let q = p;
shift(q);
print(show(p.x));
--------------------------------------------------------------------------------
1

:test-prelude

struct Point { x: i32, y: i32 }
struct Rect { min: Point, max: Point }
fn area(r: Rect) -> i32 = (r.max.x - r.min.x) * (r.max.y - r.min.y);
print(show(area(Rect { min: Point { x: 1, y: 1 }, max: Point { x: 4, y: 3 } })));
--------------------------------------------------------------------------------
6

Patterns of structs and tuples

struct Point { x: i32, y: i32 }
imm struct Code(u32);
fn describe(p: Point) -> i32 = match p {
  Point { x: 0, y } => y,
  Point { x, y: 0 } => x * 10,
  _ => -1,
};
fn sum((a, b): (i32, i32)) -> i32 = a + b;
let Code(n) = Code(5);
let (first, .., last) = (1, 2, 3, 4);
print(show(
  describe(Point { x: 0, y: 7 }) + describe(Point { x: 3, y: 0 }) + describe(Point { x: 1, y: 1 })
    + sum((first, last)) + n as i32
));
