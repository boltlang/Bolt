================================================================================
Casts between integers and floats
================================================================================
7 as f64 / 2.0;
--------------------------------------------------------------------------------
3.5

================================================================================
Casts from floats truncate
================================================================================
3.9 as i32;
--------------------------------------------------------------------------------
3

================================================================================
64-bit integers
================================================================================
fn big(n: i64) -> i64 = n * 1000000000;
big(10);
--------------------------------------------------------------------------------
10000000000

================================================================================
Shifts of 64-bit integers
================================================================================
let x: i64 = 1 << 40;
(x >> 38) as i32;
--------------------------------------------------------------------------------
4

================================================================================
Large unsigned integers
================================================================================
let x: u32 = 4000000000;
x;
--------------------------------------------------------------------------------
4000000000
