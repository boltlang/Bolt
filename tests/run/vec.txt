  let mut i = 0;
  while i < n {
    v.push("x");
    i += 1;
  }
}
let v = vec_new();
fill(v, 3);
to_array(v);
--------------------------------------------------------------------------------
["x", "x", "x"]

================================================================================
Generic functions over indexing and loops
:todo wasm
================================================================================
fn first[C](c: C) -> C.Item where Index[C] = c[0];
fn count[C](c: C) -> i32 where Iterable[C], Iterator[C.Iter] {
  let mut n = 0;
  for x in c {
    n += 1;
  }
  n
}
let v = vec_from_array(["a", "b"]);
(first(v), count(v));
--------------------------------------------------------------------------------
("a", 2)
