The operations of primitive types as builtins
================================================================================
extern "bolt" {
  fn i32_add(a: i32, b: i32) -> i32;
  fn i32_rem(a: i32, b: i32) -> i32;
  fn i32_neg(a: i32) -> i32;
  fn i64_shl(a: i64, n: u32) -> i64;
  fn u8_mul(a: u8, b: u8) -> u8;
  fn f64_div(a: f64, b: f64) -> f64;
}
(i32_add(2, 3) * 1000 + i32_rem(-7, 3) * 100 + i32_neg(4)) as i64 + i64_shl(1, 40) + (u8_mul(5, 6) as i64) + (f64_div(7.0, 2.0) as i64);
--------------------------------------------------------------------------------
1099511632705

extern "bolt" {
  fn i32_lt(a: i32, b: i32) -> Bool;
  fn f32_ge(a: f32, b: f32) -> Bool;
  fn char_eq(a: Char, b: Char) -> Bool;
  fn bool_not(a: Bool) -> Bool;
  fn bool_bitxor(a: Bool, b: Bool) -> Bool;
}
i32_lt(1, 2) && f32_ge(1.5, 1.5) && char_eq('a', 'a') && bool_not(False) && bool_bitxor(True, False);
--------------------------------------------------------------------------------
