================================================================================
Printing from functions
================================================================================
extern "bolt" { fn print(text: String); }
fn greet(name: String) {
  print("hello");
  print(name);
}
greet("Bolt");
greet("Wasm");
--------------------------------------------------------------------------------
hello
Bolt
hello
Wasm

================================================================================
Strings in variables, consts and results
================================================================================
extern "bolt" { fn print(text: String); }
const GREETING: String = "hi";
fn pick(b: Bool) -> String = if b { GREETING } else { "bye" };
let s = pick(False);
print(pick(True));
print(s);
s;
--------------------------------------------------------------------------------
hi
bye
"bye"

================================================================================
A panic
================================================================================
extern "bolt" { fn panic(message: String) -> !; fn print(text: String); }
fn check(n: i32) -> i32 = if n < 0 { panic("negative") } else { n };
print("checking");
check(1) + check(-1);
--------------------------------------------------------------------------------
checking
error: panicked: negative
