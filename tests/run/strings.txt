:prelude

impl Debug[Point] { fn debug(p, out) = out.push_str("<point>"); }
print(repr('\n'));
print(repr("a\tb"));
print(repr(Point { x: 1, y: 2 }));
--------------------------------------------------------------------------------
'\n'
"a\tb"
<point>
