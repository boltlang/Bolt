================================================================================
Tuple struct patterns
================================================================================
match x {
  Some(x) => "some",
  std.None() => "none"
}

================================================================================
Struct patterns
================================================================================
match x {
  Person{name, age} if age < 5 => ("toddler", name),
  Person{name: adult_name, age: _} => ("adult", adult_name),
}

match y {
  Bar.T1(_, Some(x)) => println!("{x}"),
}

================================================================================
Ignored patterns
================================================================================
match x {
  (a, ..) => a,
  B(..) => c,
  D.E{f: g, ..} => g
}

================================================================================
Captured patterns
================================================================================
match x {
  a @ A(_) | b @ B(..) => a,
  a @ 1 ..= 5 => a,
  Some(1 ..= 5) => a,
  a @ b..c => a,
  a @ b..=c => a,
  d.. => a,
  ..d => d,
  a @ ..=5 => a
}

match name {
  | "IPV6_FLOWINFO"
  | "IPV6_FLOWLABEL_MGR"
  | "IPV6_FLOWINFO_SEND" => True,
  _ => False,
}

================================================================================
Or patterns
================================================================================
if let A(x) | B(x) = expr {
    do_stuff_with(x);
}

while let A(x) | B(x) = expr {
    do_stuff_with(x);
}

let Ok(index) | Err(index) = slice.binary_search(x);

for a | b in c {}

let Ok(x) | Err(x) = binary_search(x);

for A | B | C in c {}

|(Ok(x) | Err(x))| expr();

let mut x @ (A | B | C);

fn foo((1 | 2 | 3): u8) {}

if let x!() | y!() = () {}

================================================================================
Inline const or Const blocks as pattern
================================================================================
fn foo(x: i32) {
    const CUBE: i32 = 3.pow(3);
    match x {
        CUBE => println!("three cubed"),
        _ => {}
    }
}

fn foo(x: i32) {
    match x {
        const { 3.pow(3) } => println!("three cubed"),
        _ => {}
    }
}

let Ok(i) | Err(i) = r;
let | A | B = c;
let 1..=5 = n;
let -5..0 = n;
let a..b = n;
let d.. = n;
let ..=d = n;
let ..d = n;
let const { 1 } = n;
let "s" = n;
let -1 = n;
