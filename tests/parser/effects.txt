================================================================================
Effect declarations
================================================================================
effect Log { fn log(message: String); }
effect State[S] {
  fn get() -> S;
  fn set(value: S);
}
pub effect Raise[E] { fn raise[T](error: E) -> T; }
effect Yield { fn yield(value: i32); }

================================================================================
Functions with effects
================================================================================
fn fetch(url: String) -> Response does Async;
fn crawl(url: String) -> () does Async + Log { visit(url); }
fn count(xs: [String]) -> i32 does State[i32] = 0;
fn is_up(url: String) does Async = check(url);
fn greet(name: String) does Log { log(name); }
fn f[T](x: T) -> T does Log where Show[T] = x;
fn double(x: i32) does Nothing = x * 2;
#[link(wasm, module = "env")]
fn log(message: String) does Console;

================================================================================
Function types with effects
================================================================================
let f: fn(A) -> B does Log = g;
let g: fn() does Async + E = h;
fn map[A, B, E](xs: [A], f: fn(A) -> B does E) -> [B] does E;
fn retry[T, E](times: i32, f: fn() -> T does E) -> T does Log + E;
let k: fn(A) -> fn(B) -> C does Log = curried;
let b: Button[Nothing] = button;

================================================================================
Closures and anonymous functions with effects
================================================================================
let a = |x| does Log { log(x) };
let b = |x| -> i32 does Log { log("x"); x };
let c = || does Async + Log { wait(t) };
let d = fn (x) does Log = log(x);

================================================================================
A closure with effects needs a block
:error
================================================================================
let a = |x| does Log log(x);

================================================================================
Handlers
================================================================================
match body() {
  () => 0,
  effect yield(x), k => x + resume(k, ()),
}
match body() {
  value => Ok(value),
  effect raise(error), _ => Err(error),
}
match body() {
  value => value,
  effect State.get(), k => resume(k, state),
  effect State.set(s), k => { state = s; resume(k, ()) },
  effect yield(0), k if quiet => resume(k, ()),
}
match body() {
  v => v,
  effect info(m) | warn(m), k => { print(m); resume(k, ()) },
  effect raise(e) | State.fail(e), _ => Err(e),
}

================================================================================
An effect arm needs a continuation
:error
================================================================================
match body() {
  () => 0,
  effect yield(x) => x,
}

================================================================================
An effect arm needs the parentheses of its operation
:error
================================================================================
match body() {
  v => v,
  effect get, k => resume(k, 0),
}

================================================================================
Every operation in an or-pattern needs its parentheses
:error
================================================================================
match body() {
  v => v,
  effect get() | reset, k => resume(k, 0),
}
