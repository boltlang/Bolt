================================================================================
Assigning to a local, a field and an index
:test-prelude
================================================================================
struct P { x: i32 }
fn f(xs: mut [i32]) {
  let mut n = 1;
  n = 2;
  let p = mut P { x: 1 };
  p.x = 3;
  xs[0] = n;
}
--------------------------------------------------------------------------------
fn f(xs: mut [i32]) -> () = {
  let mut n = 1;
  n = 2;
  let p = P(1);
  p.P.0 = 3;
  xs[0] = n with IndexMut[mut [i32]].set_index;
}

================================================================================
Compound assignments
:test-prelude
================================================================================
fn f(xs: mut [i32]) {
  let mut n = 1;
  n += 2;
  xs[0] += n;
}
--------------------------------------------------------------------------------
fn f(xs: mut [i32]) -> () = {
  let mut n = 1;
  n += 2;
  xs[0] += n with Index[mut [i32]].index, IndexMut[mut [i32]].set_index;
}
