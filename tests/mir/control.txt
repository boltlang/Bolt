================================================================================
Patterns in let, if let and let else
:test-prelude
================================================================================
fn f(o: Option[i32], p: (i32, i32)) -> i32 {
  let (a, b) = p;
  let _ = a;
  if let Some(x) = o {
    return x;
  }
  let Some(y) = o else { return b };
  y
}
--------------------------------------------------------------------------------
fn f(o: Option[i32], p: (i32, i32)) -> i32 = {
  let {
    $1 = $0.0
    $2 = $0.1
    n0 = arm 0 with a = $1, b = $2
  } = p;
  let _ = a;
  if let {
    $1 = $0.Some.0
    n0 = switch $0 { Some => n1, _ => n2 }
    n1 = arm 0 with x = $1
    n2 = fail
  } = o {
    return x;
  };
  let {
    $1 = $0.Some.0
    n0 = switch $0 { Some => n1, _ => n2 }
    n1 = arm 0 with y = $1
    n2 = fail
  } = o else {
    return b
  };
  y
}
