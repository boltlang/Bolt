fn f(n: i32) {
  let mut total = 0;
  let add = |x| {
    total = total + x + n;
  };
  add(1);
}
--------------------------------------------------------------------------------
fn f(n: i32) -> () = {
  let mut total = 0;
  let add = lambda#9 capturing [ref total, n];
  add(1);
}

  total = `+`(`+`(total, x), n);
