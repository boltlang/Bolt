================================================================================
A function with operators
:test-prelude
================================================================================
fn fac(x: i32) -> i32 {
  if x == 0 {
    return 1
  } else {
    return x * fac(x - 1);
  }
}
--------------------------------------------------------------------------------
fn fac(x: i32) -> i32 = {
  if `==`(x, 0) {
    return 1
  } else {
    return `*`(x, fac(`-`(x, 1)));
  }
}
