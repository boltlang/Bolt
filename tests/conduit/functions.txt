================================================================================
Recursion
:test-prelude
================================================================================
extern "env" { fn log(x: i32); }
fn fac(n: i32) -> i32 = if n <= 1 { 1 } else { n * fac(n - 1) };
log(fac(5));
--------------------------------------------------------------------------------
extern log "env" "log" (i32) -> ()

fn file() -> () {
  let %0: i32 = call fac(5i32)
  let = call_extern log(%0)
  jump ret()
}

fn fac(%n: i32) -> (i32) {
  join j1(%1: i32) = jump ret(%1)
  in
  let %2: i32 = call lt_eq_i32(%n, 1i32)
  case %2 {
    0 =>
      let %3: i32 = call minus_i32(%n, 1i32)
      let %4: i32 = call fac(%3)
      let %5: i32 = call star_i32(%n, %4)
      jump j1(%5)
    _ => jump j1(1i32)
  }
}

fn lt_eq_i32(%a: i32, %b: i32) -> (i32) {
  let %2: i32 = call PartialOrd_i32_lte(%a, %b)
  jump ret(%2)
}

fn minus_i32(%a: i32, %b: i32) -> (i32) {
  let %2: i32 = call Sub_i32_sub(%a, %b)
  jump ret(%2)
}

fn star_i32(%a: i32, %b: i32) -> (i32) {
  let %2: i32 = call Mul_i32_mul(%a, %b)
  jump ret(%2)
}

fn PartialOrd_i32_lte(%a: i32, %b: i32) -> (i32) {
  let %2: i32 = le.s(%a, %b)
  jump ret(%2)
}

fn Sub_i32_sub(%a: i32, %b: i32) -> (i32) {
  let %2: i32 = sub(%a, %b)
  jump ret(%2)
}

fn Mul_i32_mul(%a: i32, %b: i32) -> (i32) {

  jump ret(%2)
