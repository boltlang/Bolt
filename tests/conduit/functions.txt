fn fac(n: i32) -> i32 = if n <= 1 { 1 } else { n * fac(n - 1) };

// fn5: *[i32]
// fn6: PartialOrd[i32].lte
// fn7: Sub[i32].sub
// fn8: Mul[i32].mul

extern extern1 "env" "log" (i32) -> ()

  let %0: i32 = call fn2(5i32)
  let = call_extern extern1(%0)
  jump ret()

fn fn2(%0: i32) -> (i32) {
  join j1(%1: i32) = jump ret(%1)
  in
  let %2: i32 = call fn3(%0, 1i32)
  case %2 {
    0 =>
      let %3: i32 = call fn4(%0, 1i32)
      let %4: i32 = call fn2(%3)
      let %5: i32 = call fn5(%0, %4)
      jump j1(%5)
    _ => jump j1(1i32)
  }
}

fn fn3(%0: i32, %1: i32) -> (i32) {
  let %2: i32 = call fn6(%0, %1)
