================================================================================
A lambda captures the locals it uses
:test-prelude
================================================================================
let n = 1;
let add = |x| x + n;
--------------------------------------------------------------------------------
`|x| x + n` captures n

================================================================================
Parameters and the lambda's own locals aren't captured
:test-prelude
================================================================================
let k = 2;
let f = |x| {
  let y = x * k;
  y + 1
};
--------------------------------------------------------------------------------
`|x| {
  let y = x * k;
  y + 1
}` captures k

================================================================================
A lambda captures each local once, in the order it first uses them
:test-prelude
================================================================================
let a = 1;
let b = 2;
let f = || b + a + b;
--------------------------------------------------------------------------------
`|| b + a + b` captures b, a

================================================================================
A local that's assigned is captured too
:test-prelude
================================================================================
let mut count = 0;
let tick = || { count += 1; };
--------------------------------------------------------------------------------
`|| { count += 1; }` captures count

================================================================================
A lambda in a lambda makes the outer one capture what it uses from outside
:test-prelude
================================================================================
let n = 1;
let outer = |x| {
  let inner = |y| y + x + n;
  inner(x)
};
--------------------------------------------------------------------------------
`|x| {
  let inner = |y| y + x + n;
  inner(x)
}` captures n
`|y| y + x + n` captures x, n

================================================================================
Lambdas in functions capture their parameters
:test-prelude
================================================================================
fn adder(n: i32) -> fn(i32) -> i32 = |x| x + n;
--------------------------------------------------------------------------------
`|x| x + n` captures n

================================================================================
A lambda that uses nothing from outside
================================================================================
let f = |x| x;
--------------------------------------------------------------------------------
`|x| x` captures nothing
