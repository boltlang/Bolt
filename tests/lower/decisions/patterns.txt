================================================================================
A switch on variants binds their fields
:test-prelude
================================================================================
let x = Option.Some(1);
match x { Option.Some(y) => y, Option.None => 0 };
--------------------------------------------------------------------------------
match `x`
$1 = $0.Some.0
n0 = switch $0 { Some => n1, None => n2 }
n1 = arm 0 with b1 = $1
n2 = arm 1

================================================================================
A match that misses a variant fails
:test-prelude
================================================================================
let x = Option.Some(1);
match x { Option.Some(1) => 1, Option.Some(_) => 2 };
--------------------------------------------------------------------------------
match `x`
$1 = $0.Some.0
n0 = switch $0 { Some => n1, _ => n4 }
n1 = switch $1 { 1 => n2, _ => n3 }
n2 = arm 0
n3 = arm 1
n4 = fail
missing Option.None

================================================================================
Tuples test one place at a time
================================================================================
let p = (True, 1);
match p { (True, n) => n, (False, 0) => 0, (_, _) => 2 };
--------------------------------------------------------------------------------
match `p`
$1 = $0.0
$2 = $0.1
n0 = switch $1 { true => n1, false => n2 }
n1 = arm 0 with b2 = $2
n2 = switch $2 { 0 => n3, _ => n4 }
n3 = arm 1
n4 = arm 2

================================================================================
Strings are compared one at a time
================================================================================
let s = "a";
match s { "a" => 1, "b" => 2, _ => 3 };
--------------------------------------------------------------------------------
match `s`
n0 = if $0 == "a" then n1 else n2
n1 = arm 0
n2 = if $0 == "b" then n3 else n4
n3 = arm 1
n4 = arm 2

================================================================================
Ranges and arms never taken
================================================================================
let n = 5;
match n { 0 => 0, 1..=9 => 1, _ => 2, 3 => 3 };
--------------------------------------------------------------------------------
match `n`
n0 = switch $0 { 0 => n1, _ => n2 }
n1 = arm 0
n2 = if $0 in 1..=9 then n3 else n4
n3 = arm 1
n4 = arm 2
unreached 3

================================================================================
A guard falls through to the next arm
:test-prelude
================================================================================
let x = Option.Some(1);
match x { Option.Some(y) if y > 0 => y, _ => 0 };
--------------------------------------------------------------------------------
match `x`
$1 = $0.Some.0
n0 = switch $0 { Some => n1, _ => n3 }
n1 = guard 0 with b1 = $1 then n2 else n3
n2 = arm 0 with b1 = $1
n3 = arm 1

================================================================================
An or pattern shares its arm
:test-prelude
================================================================================
let x = (Option.Some(1), Option.Some(2));
match x { (Option.Some(a), Option.Some(b)) => a + b, (Option.None, _) | (_, Option.None) => 0 };
--------------------------------------------------------------------------------
match `x`
$1 = $0.0
$2 = $0.1
$3 = $1.Some.0
$4 = $2.Some.0
n0 = switch $1 { Some => n1, None => n3 }
n1 = switch $2 { Some => n2, None => n3 }
n2 = arm 0 with b1 = $3, b2 = $4
n3 = arm 1

================================================================================
Slice patterns are not supported yet
:error
================================================================================
let xs = [1];
match xs { [a] => a, _ => 0 };
--------------------------------------------------------------------------------
match `xs`
error: the pattern `[a]` isn't supported yet

================================================================================
Each match of each file is shown
================================================================================
//- /lib.bolt
pub fn sign(n: i32) -> i32 = match n { 0 => 0, _ => 1 };
//- /main.bolt
import "./lib" (sign);
match sign(2) { 0 => False, _ => True };
--------------------------------------------------------------------------------
/lib.bolt: match `n`
n0 = switch $0 { 0 => n1, _ => n2 }
n1 = arm 0
n2 = arm 1
/main.bolt: match `sign(2)`
n0 = switch $0 { 0 => n1, _ => n2 }
n1 = arm 0
n2 = arm 1

================================================================================
A variant pattern with the wrong number of fields has no decision
:error
================================================================================
enum Shape { Circle(f64), Rect(f64, f64) }
fn f(s: Shape) = match s {
  Shape.Circle(r, extra) => 1,
  Shape.Rect(w) => 2,
};
--------------------------------------------------------------------------------
match `s`
error: the pattern `Shape.Circle(r, extra)` isn't known
