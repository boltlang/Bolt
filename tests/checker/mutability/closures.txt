================================================================================
An imm struct can't hold a closure that changes what it captures (#244)
:error
================================================================================

struct Counter { n: i32 }
let c = mut Counter { n: 0 };

--------------------------------------------------------------------------------
let c: mut Counter
let h: imm Holder
--------------------------------------------------------------------------------
error at `|| { c.n = 1; }`: expected fn() -> (), found mut fn() -> ()
