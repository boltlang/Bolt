================================================================================
Mut before a literal makes it mut
:prelude
================================================================================
struct Point { x: i32 }
struct Wrapper(i32);
let a = mut [1, 2];
let b = mut [[1]];
let c = mut [0; 3];
let p = mut Point { x: 1 };
let w = mut Wrapper(1);
a[0] = 3;
p.x = 2;
--------------------------------------------------------------------------------
let a: mut [i32]
let b: mut [imm [i32]]
let c: mut [i32]
let p: mut Point
let w: mut Wrapper
a[0] = 3: ()
p.x = 2: ()

================================================================================
Mut before anything else is an error
:error
:prelude
================================================================================
fn f(x: i32) -> [i32] = [x];
let xs = [1, 2];
let a = mut xs;
let b = mut f(1);
let c = mut "abc";
let d = mut ([1], 2);
fn g[T](x: T) -> T = mut x;
--------------------------------------------------------------------------------
f: fn(i32) -> imm [i32]
g: forall T. fn(T) -> T
let xs: imm [i32]
let a: imm [i32]
let b: imm [i32]
let c: String
let d: (imm [i32], i32)
--------------------------------------------------------------------------------
error at `mut x`: can't be mut
error at `mut xs`: can't be mut
error at `mut f(1)`: can't be mut
error at `mut "abc"`: can't be mut
error at `mut ([1], 2)`: can't be mut
