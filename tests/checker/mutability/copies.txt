================================================================================
A deep copy that can't be changed can't be changed inside either
:error
:test-prelude
================================================================================
let a = to_imm_deep(mut [mut [1, 2]]);
a[0][0] = 3;
--------------------------------------------------------------------------------
let a: imm [imm [i32]]
a[0][0] = 3: ()
--------------------------------------------------------------------------------
error at `a[0]`: can't change imm [i32]

================================================================================
A function has no deep copy
:error
:test-prelude
================================================================================
let f = to_imm_deep(|x: i32| x);
--------------------------------------------------------------------------------
let f: fn(x: i32) -> i32
--------------------------------------------------------------------------------
error at `to_imm_deep`: no instance ToImmDeep[fn(x: i32) -> i32]

================================================================================
