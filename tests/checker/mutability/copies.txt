================================================================================
The types of copies
:prelude
================================================================================
let a = to_mut([1]);
let b = to_imm(mut [1]);
let c = to_mut(Vec.from_array([True]));
let d = to_imm(Vec.from_array([True]));
--------------------------------------------------------------------------------
let a: mut [i32]
let b: imm [i32]
let c: mut Vec[Bool]
let d: imm Vec[Bool]

================================================================================
A copy that can't be changed can't be changed
:error
:prelude
================================================================================
let xs = to_imm(mut [1]);
xs[0] = 2;
let v = to_imm(Vec.from_array([1]));
v.push(2);
--------------------------------------------------------------------------------
let xs: imm [i32]
xs[0] = 2: ()
let v: imm Vec[i32]
v.push(2): ()
--------------------------------------------------------------------------------
error at `xs`: can't change imm [i32]
error at `v`: expected mut Vec[{integer}], found imm Vec[{integer}]

================================================================================
An imm copy can't hold anything mut
:error
:prelude
================================================================================
struct Point { x: i32 }
let points = to_imm(mut [mut Point { x: 1 }]);
--------------------------------------------------------------------------------
let points: imm [mut Point]
--------------------------------------------------------------------------------
error at `to_imm`: mut Point isn't at most imm

================================================================================
Types without the impls
:error
:prelude
================================================================================
struct Point { x: i32 }
to_mut(Point { x: 1 });
to_imm(1);
--------------------------------------------------------------------------------
to_mut(Point { x: 1 }): {error}
to_imm(1): {error}
--------------------------------------------------------------------------------
error at `to_mut`: no instance ToMut[imm Point]
error at `to_imm`: no instance ToImm[i32]

================================================================================
The types of deep copies that can't be changed
:prelude
================================================================================
let a = to_imm_deep(mut [mut [1, 2], mut [3]]);
let v = Vec.new();
v.push(mut [1]);
let b = to_imm_deep(v);
let c = to_imm_deep(Some(mut [True]));
let d = to_imm_deep(mut (mut [1], 'x', "s"));
let e = to_imm_deep(List.Cons(mut [1], List.Nil));
let f = to_imm_deep(5);
--------------------------------------------------------------------------------
let a: imm [imm [i32]]
let v: mut Vec[mut [i32]]
v.push(mut [1]): ()
let b: imm Vec[imm [i32]]
let c: Option[imm [Bool]]
let d: imm (imm [i32], Char, String)
let e: List[imm [i32]]
let f: i32

================================================================================
A deep copy that can't be changed can't be changed inside either
:error
:prelude
================================================================================
let a = to_imm_deep(mut [mut [1, 2]]);
a[0][0] = 3;
--------------------------------------------------------------------------------
let a: imm [imm [i32]]
a[0][0] = 3: ()
--------------------------------------------------------------------------------
error at `a[0]`: can't change imm [i32]

================================================================================
A function has no deep copy
:error
:prelude
================================================================================
let f = to_imm_deep(|x: i32| x);
--------------------------------------------------------------------------------
let f: fn(x: i32) -> i32
--------------------------------------------------------------------------------
error at `to_imm_deep`: no instance ToImmDeep[fn(x: i32) -> i32]

let b: imm (i32, i32, i32, i32, i32, i32, i32, i32, i32, imm [i32])

================================================================================
A deep copy must be imm all the way down
:prelude
:error
================================================================================

impl ToImmDeep[Cell] { fn to_imm_deep(c) = c; }
--------------------------------------------------------------------------------

--------------------------------------------------------------------------------
