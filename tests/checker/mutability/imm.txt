================================================================================
Imm of a value built step by step
:prelude
================================================================================
let ys = imm {
  let xs = mut [0; 3];
  xs[1] = 2;
  xs
};
let v = imm {
  let v = Vec.new();
  v.push(1);
  v
};
--------------------------------------------------------------------------------
let ys: imm [i32]
let v: imm Vec[i32]

================================================================================
Imm is deep
================================================================================
struct Point { x: i32 }
struct Line { a: Point, b: Point }
let ps = imm mut [mut Point { x: 1 }];
let l = imm mut Line { a: mut Point { x: 1 }, b: mut Point { x: 2 } };
let t = imm (mut [1], 2);
let n = imm 1;
--------------------------------------------------------------------------------
let ps: imm [imm Point]
let l: imm Line
let t: (imm [i32], i32)
let n: i32

================================================================================
Imm of a struct gives its type arguments what imm makes of its fields
:test-prelude
================================================================================
struct Box[T] { item: T }
struct Handler[T] { f: imm fn(T) -> () }
let b = imm mut Box { item: mut [1] };
let h = imm mut Handler { f: |xs: mut [i32]| { xs[0] = 1; } };
--------------------------------------------------------------------------------
let b: imm Box[imm [i32]]
let h: imm Handler[mut [i32]]

================================================================================
Imm of a struct that holds itself
================================================================================
struct Node { value: i32, next: Option[Node] }
enum Option[T] { Some(T), None }
let n = imm mut Node { value: 1, next: Option.None };
--------------------------------------------------------------------------------
let n: imm Node

================================================================================
Imm of a value whose type is found later
================================================================================
let f = |x| imm x;
let ys = f(mut [1]);
--------------------------------------------------------------------------------
let f: fn(mut [i32]) -> imm [i32]
let ys: imm [i32]

================================================================================
An imm copy can't be changed
:error
:test-prelude
================================================================================
let xs = mut [1];
let ys = imm xs;
ys[0] = 2;
--------------------------------------------------------------------------------
let xs: mut [i32]
let ys: imm [i32]
ys[0] = 2: ()
--------------------------------------------------------------------------------
error at `ys`: can't change imm [i32]

================================================================================
A closure that changes something can't be imm
:error
:test-prelude
================================================================================
let xs = mut [0];
let f = imm (|| { xs[0] = 1; });
--------------------------------------------------------------------------------
let xs: mut [i32]
let f: mut fn() -> ()
--------------------------------------------------------------------------------
error at `imm (|| { xs[0] = 1; })`: mut fn() -> () isn't at most imm

================================================================================
Imm of a type parameter needs it to be imm
:error
================================================================================
fn f[T](x: T) -> T = imm x;
fn g[T: imm](x: T) -> T = imm x;
fn h(x) = imm x;
--------------------------------------------------------------------------------
f: forall T. fn(T) -> T
g: forall T. T: imm => fn(T) -> T
h: forall a. a: imm => fn(a) -> a
--------------------------------------------------------------------------------
error at `imm x`: T isn't at most imm

================================================================================
A type argument held and in a function type must already be imm
:error
================================================================================
struct Both[T] { x: T, f: imm fn(T) -> () }
let a = imm mut Both { x: [1], f: |_| {} };
let b = imm mut Both { x: mut [1], f: |_| {} };
--------------------------------------------------------------------------------
let a: imm Both[imm [i32]]
let b: imm Both[imm [i32]]
--------------------------------------------------------------------------------
error at `imm mut Both { x: mut [1], f: |_| {} }`: expected fn(imm [{integer}]) -> (), found fn(mut [{integer}]) -> ()
