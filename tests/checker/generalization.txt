================================================================================
A literal in a function without a signature defaults when generalized
================================================================================
fn inc(x) = x + 1;
inc(1u8);

--------------------------------------------------------------------------------
inc: fn(i32) -> i32
inc(1u8): i32
--------------------------------------------------------------------------------
error at `1u8`: expected i32, found u8

================================================================================
A function that returns a literal can't be used at another type
================================================================================
fn zero() = 0;
let a: u8 = zero();

--------------------------------------------------------------------------------
zero: fn() -> i32
let a: u8
--------------------------------------------------------------------------------
error at `zero()`: expected u8, found i32

================================================================================
An empty array from a function is generic
================================================================================
fn empty() = [];
let a: [i32] = empty();
let b: [Bool] = empty();

--------------------------------------------------------------------------------
empty: forall a. fn() -> [a]
let a: [i32]
let b: [Bool]

================================================================================
A None from a function is generic
================================================================================
enum Option[T] { None, Some(T) }
fn none() = Option.None;
let a: Option[i32] = none();
let b: Option[String] = none();

--------------------------------------------------------------------------------
none: forall a. fn() -> Option[a]
let a: Option[i32]
let b: Option[String]

================================================================================
Top-level lets are not generic
================================================================================
let id = |x| x;
id(1);
id(True);

--------------------------------------------------------------------------------
let id: fn(i32) -> i32
id(1): i32
id(True): i32
--------------------------------------------------------------------------------
error at `True`: expected {integer}, found Bool

================================================================================
Functions can't see top-level lets
================================================================================
let n = 1;
fn f() = n + 1;
fn g() -> Bool = n;

--------------------------------------------------------------------------------
f: fn() -> {error}
g: fn() -> Bool
let n: i32
--------------------------------------------------------------------------------
error at `n`: unresolved
error at `n`: unresolved

================================================================================
Shadowing gives a new type
================================================================================
let x = 1;
let x = "a";
x;
fn f(y: i32) = { let y = True; y };

--------------------------------------------------------------------------------
f: fn(i32) -> Bool
let x: i32
let x: String
x: String

================================================================================
Nested lambdas
================================================================================
let add = |a| |b| a + b;
add(1)(2);
(|x| x)(True);

--------------------------------------------------------------------------------
let add: fn(i32) -> fn(i32) -> i32
add(1)(2): i32
(|x| x)(True): Bool

================================================================================
Lambdas with patterns as parameters
================================================================================
let sum = |(a, b)| a + b;
sum((1, 2));
sum(1);

--------------------------------------------------------------------------------
let sum: fn((i32, i32)) -> i32
sum((1, 2)): i32
sum(1): i32
--------------------------------------------------------------------------------
error at `1`: expected ({integer}, {integer}), found {integer}

================================================================================
Class constraints in lambdas are inferred for their function
================================================================================
class Show[T] { fn show(x: T) -> String; }
fn f(x) = (|y| show(y))(x);
fn g(x) { let s = |y| show(y); s(x) }

--------------------------------------------------------------------------------
f: forall a. Show[a] => fn(a) -> String
g: forall a. Show[a] => fn(a) -> String

================================================================================
Generic functions used at several types in one function
================================================================================
class Show[T] { fn show(x: T) -> String; }
impl Show[i32] { fn show(x) = "int"; }
impl Show[Bool] { fn show(x) = "Bool"; }
fn both() = (show(1), show(True));
fn twice(x) = (show(x), show(x));
fn mixed(x) = (twice(x), twice(1));

--------------------------------------------------------------------------------
Show[i32].show: fn(i32) -> String
Show[Bool].show: fn(Bool) -> String
both: fn() -> (String, String)
twice: forall a. Show[a] => fn(a) -> (String, String)
mixed: forall a. Show[a] => fn(a) -> ((String, String), (String, String))
