================================================================================
A superclass's associated type
================================================================================
class Index[C] {
  type Item;
  fn get(c: C, i: i32) -> Item;
}
class IndexMut[C] where Index[C] {
  fn set(c: C, i: i32, x: C.Item);
}
struct Box[T] { items: [T] }
impl[T] Index[Box[T]] {
  type Item = T;
  fn get(c, i) = c.items[i];
}
impl[T] IndexMut[mut Box[T]] {
  fn set(c, i, x) { c.items[i] = x; }
}
fn put[C](c: C, x: C.Item) where IndexMut[C] = set(c, 0, x);
