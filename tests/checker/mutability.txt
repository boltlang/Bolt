================================================================================
Assigning to fields and elements
================================================================================
struct Point { x: i32, y: i32 }
fn f() {
  let p = mut Point { x: 1, y: 2 };
  p.x = 3;
  p.y = True;
  let a = mut [1, 2];
  a[0] = 3;
  a[1] = "b";
}

--------------------------------------------------------------------------------
f: fn() -> ()
--------------------------------------------------------------------------------
error at `True`: expected i32, found Bool
error at `"b"`: expected {integer}, found String

}

================================================================================
A closure can change a mut it captures
================================================================================
fn f() -> i32 {
  let mut count = 0;
  let tick = || { count += 1; };
  tick();
  count
}

--------------------------------------------------------------------------------
f: fn() -> i32

================================================================================
A mut parameter can be assigned
================================================================================

}
