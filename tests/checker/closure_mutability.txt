================================================================================
A mut struct can hold a mut closure
================================================================================
struct Holder { poke: fn() }
let xs = mut [0];
let h = mut Holder { poke: || { xs[0] += 1; } };
h.poke();

--------------------------------------------------------------------------------
let xs: mut [i32]
let h: mut Holder
h.poke(): ()

================================================================================
A closure that captures nothing that can change is imm
================================================================================
struct Holder { get: fn() -> i32 }
let k = 1;
let xs = [1, 2];
let h = Holder { get: || k + xs[0] };

--------------------------------------------------------------------------------
let k: i32
let xs: imm [i32]
let h: imm Holder

================================================================================
Calling a mut closure needs nothing mut
================================================================================
fn apply_twice(f, x) = f(f(x));
let mut total = 0;
apply_twice(|x| { total += x; x }, 1);
apply_twice(|x| x + 1, 1);

apply_twice(|x| { total += x; x }, 1): i32
