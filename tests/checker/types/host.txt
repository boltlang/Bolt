fn set_text(e: Element, s: JsString);
fn copy(s: JsString) -> JsString = imm s;
fn twice(e: Element) = (e, imm e);

================================================================================
A type of the host has functions in its namespace
================================================================================
//- /js.bolt
#[link(wasm)]
pub type JsString;
#[link(wasm, module = "bolt:js", name = "make")]
fn make(n: i32) -> JsString;
pub fn JsString.of(n: i32) -> JsString = make(n);
//- /main.bolt
import "./js" (JsString);
let s = JsString.of(1);
--------------------------------------------------------------------------------

/js.bolt: JsString.of: fn(i32) -> JsString
