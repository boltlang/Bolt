================================================================================
Keyword-only parameters
================================================================================
fn split(s: i32, *, at: i32) -> i32 = s;
let a = split(1, at: 2);
let f: fn(i32, *, at: i32) -> i32 = split;
let b = f(3, at: 4);
--------------------------------------------------------------------------------
split: fn(i32, *, at: i32) -> i32
let a: i32
let f: fn(i32, *, at: i32) -> i32
let b: i32

================================================================================
Rest parameters
================================================================================
fn count[*Ts](*args: *Ts) -> (*Ts) = args;
fn opts[**Ks](x: i32, **o: **Ks) -> (**Ks) = o;
fn both[*Ts, **Ks](*args: *Ts, sep: i32, **o: **Ks) -> i32 = sep;
let a = count(1, True, 2u8);
let b = count();
let c = opts(1, y: True, z: 3);
let d = opts(1);
let e = both(1, 2, sep: 3, w: True);
--------------------------------------------------------------------------------
count: forall Ts. fn(*Ts) -> (*Ts)
opts: forall Ks. fn(i32, **Ks) -> (**Ks)
both: forall Ts Ks. fn(*Ts, sep: i32, **Ks) -> i32
let a: (i32, Bool, u8)
let b: ()
let c: (y: Bool, z: i32)
let d: ()
let e: i32

================================================================================
Spreads in calls
================================================================================
fn count[*Ts](*args: *Ts) -> (*Ts) = args;
fn opts[**Ks](x: i32, **o: **Ks) -> (**Ks) = o;
let t = (1, True);
let a = count(*t, 3);
let r = (k: 1);
let b = opts(0, **r, z: True);
--------------------------------------------------------------------------------
count: forall Ts. fn(*Ts) -> (*Ts)
opts: forall Ks. fn(i32, **Ks) -> (**Ks)
let t: imm (i32, Bool)
let a: (i32, Bool, i32)
let r: imm (k: i32)
let b: (k: i32, z: Bool)

================================================================================
Keyword parameters are in the type of a function, in the order of their names
================================================================================
fn order(*, z: i32, a: Bool) -> i32 = z;
let a = order(a: True, z: 1);
--------------------------------------------------------------------------------
order: fn(*, a: Bool, z: i32) -> i32
let a: i32

================================================================================
Passing a pack on
================================================================================
fn count[*Ts](*args: *Ts) -> (*Ts) = args;
fn fwd[*Ts](*args: *Ts) -> (*Ts) = count(*args);
--------------------------------------------------------------------------------
count: forall Ts. fn(*Ts) -> (*Ts)
fwd: forall Ts a. fn(*Ts) -> (*Ts)
