================================================================================
A tuple with named fields
================================================================================
let p = (x: 1, y: True);
let a = p.x;
let b = p.y;
let d = (y: 1, x: 2);
--------------------------------------------------------------------------------
let p: imm (x: i32, y: Bool)
let a: i32
let b: Bool
let d: imm (x: i32, y: i32)

================================================================================
The order of named fields doesn't matter
================================================================================
fn get_x(p: (x: i32, y: Bool)) -> i32 = p.x;
let a = get_x((y: True, x: 4));
--------------------------------------------------------------------------------
get_x: fn(p: (x: i32, y: Bool)) -> i32
let a: i32

================================================================================
Spreads
================================================================================
let p = (x: 1, y: 2);
let t = (True, 2u8);
let a = (z: 3, **p);
let b = (*t, 1);
let c = (0, *t, *t);
let d = (**p, w: 1);
--------------------------------------------------------------------------------
let p: imm (x: i32, y: i32)
let t: imm (Bool, u8)
let a: imm (x: i32, y: i32, z: i32)
let b: imm (Bool, u8, i32)
let c: imm (i32, Bool, u8, Bool, u8)
let d: imm (w: i32, x: i32, y: i32)

================================================================================
A spread waits for its type
================================================================================
fn f(t: (x: i32)) = {
  let u = (**v, y: 1);
  let v = t;
  u
};
--------------------------------------------------------------------------------
f: fn(t: (x: i32)) -> {error}
--------------------------------------------------------------------------------
error at `v`: unresolved

================================================================================
A spread of a parameter whose type is known later
================================================================================
fn wrap(t) = (**t, y: 1);
--------------------------------------------------------------------------------
wrap: forall a. fn(t: a) -> {error}
--------------------------------------------------------------------------------
error at `(**t, y: 1)`: type needed

================================================================================
A field given twice is an error
:error
================================================================================
let p = (x: 1);
let a = (x: 1, x: 2);
let b = (x: 2, **p);
fn f(t: (x: i32, x: Bool)) = t;
--------------------------------------------------------------------------------
f: fn(t: (x: i32)) -> (x: i32)
let p: imm (x: i32)
let a: imm (x: i32)
let b: imm (x: i32)
--------------------------------------------------------------------------------
error at `(x: i32, x: Bool)`: x is given twice
error at `(x: 1, x: 2)`: x is given twice
error at `(x: 2, **p)`: x is given twice

================================================================================
Only a tuple of its kind can be spread
:error
================================================================================
let p = (x: 1);
let t = (1, 2);
let a = (*1);
let b = (*p);
let c = (**t);
let d = (**());
--------------------------------------------------------------------------------
let p: imm (x: i32)
let t: imm (i32, i32)
let a: {error}
let b: {error}
let c: {error}
let d: ()
--------------------------------------------------------------------------------
error at `(*p)`: imm (x: i32) can't be spread
error at `(**t)`: imm (i32, i32) can't be spread
error at `(*1)`: i32 can't be spread

================================================================================
Tuples with named fields can be mut
================================================================================
let m = mut (x: 1, y: [1]);
m.x = 2;
--------------------------------------------------------------------------------
let m: mut (x: i32, y: imm [i32])
m.x = 2: ()

================================================================================
A field of a tuple that isn't mut can't be assigned
:error
================================================================================
let m = (x: 1);
m.x = 2;
--------------------------------------------------------------------------------
let m: imm (x: i32)
m.x = 2: ()
--------------------------------------------------------------------------------
error at `m`: can't change imm (x: i32)

================================================================================
Patterns with named fields
================================================================================
let p = (x: 2, y: True);
let a = match p { (x: 0, y: _) => 0, (x: n, ..) => n };
let (y: flag, ..) = p;
--------------------------------------------------------------------------------
let p: imm (x: i32, y: Bool)
let a: i32
let (y: flag, ..): imm (x: i32, y: Bool)

================================================================================
A pattern without .. names every field
:error
================================================================================
let p = (x: 1, y: 2);
let (x: a) = p;
--------------------------------------------------------------------------------
let p: imm (x: i32, y: i32)
let (x: a): imm (x: i32, y: i32)
--------------------------------------------------------------------------------
error at `(x: a)`: expected imm (x: {integer}, y: {integer}), found (x: a)

================================================================================
A pattern names fields the type has
:error
================================================================================
let p = (x: 1, y: 2);
let (z: a, ..) = p;
--------------------------------------------------------------------------------
let p: imm (x: i32, y: i32)
let (z: a, ..): imm (x: i32, y: i32)
--------------------------------------------------------------------------------
error at `(z: a, ..)`: imm (x: i32, y: i32) has no field z

================================================================================
A pattern with .. waits for its type
================================================================================
let f = |s| { let (x: b, ..) = s; b };
let c = f((x: 1, y: True));
--------------------------------------------------------------------------------
let f: fn(s: imm (x: i32, y: Bool)) -> i32
let c: i32

================================================================================
A pattern with .. whose type is never known
================================================================================
fn first(t) = { let (x: b, ..) = t; b };
--------------------------------------------------------------------------------
first: forall a. fn(t: a) -> {error}
--------------------------------------------------------------------------------
error at `(x: b, ..)`: type needed
