================================================================================
A tuple with named fields
================================================================================
let p = (x: 1, y: True);
let a = p.x;
let b = p.y;
let q = (1, z: True);
let c = q.0;
let d = (y: 1, x: 2);
--------------------------------------------------------------------------------
let p: imm (x: i32, y: Bool)
let a: i32
let b: Bool
let q: imm (i32, z: Bool)
let c: i32
let d: imm (x: i32, y: i32)

================================================================================
The order of named fields doesn't matter
================================================================================
fn get_x(p: (x: i32, y: Bool)) -> i32 = p.x;
let a = get_x((y: True, x: 4));
--------------------------------------------------------------------------------
get_x: fn((x: i32, y: Bool)) -> i32
let a: i32

================================================================================
Spreads
================================================================================
let p = (x: 1, y: 2);
let t = (True, 'a');
let a = (z: 3, ...p);
let b = (*t, 1, n: 2);
let c = (0, ...t, **p);
let d = (*p, w: 1);
--------------------------------------------------------------------------------
let p: imm (x: i32, y: i32)
let t: imm (Bool, {error})
let a: imm (x: i32, y: i32, z: i32)
let b: imm (Bool, {error}, i32, n: i32)
let c: imm (i32, Bool, {error}, x: i32, y: i32)
let d: imm (w: i32)
--------------------------------------------------------------------------------
error at `'a'`: needs Char from the prelude

================================================================================
A spread waits for its type
================================================================================
fn f(t: (i32, x: i32)) = {
  let u = (...v, y: 1);
  let v = t;
  u
};
--------------------------------------------------------------------------------
f: fn((i32, x: i32)) -> {error}
--------------------------------------------------------------------------------
error at `v`: unresolved

================================================================================
A spread of a parameter whose type is known later
================================================================================
fn wrap(t) = (...t, y: 1);
--------------------------------------------------------------------------------
wrap: forall a b. fn(a) -> b
--------------------------------------------------------------------------------
error at `(...t, y: 1)`: type needed

================================================================================
A field given twice is an error
:error
================================================================================
let p = (x: 1);
let a = (x: 1, x: 2);
let b = (x: 2, ...p);
fn f(t: (x: i32, x: Bool)) = t;
--------------------------------------------------------------------------------
f: fn((x: i32)) -> (x: i32)
let p: imm (x: i32)
let a: imm (x: i32)
let b: imm (x: i32)
--------------------------------------------------------------------------------
error at `(x: i32, x: Bool)`: x is given twice
error at `(x: 1, x: 2)`: x is given twice
error at `(x: 2, ...p)`: x is given twice

================================================================================
Only a tuple can be spread
:error
================================================================================
let a = (...1);
--------------------------------------------------------------------------------
let a: {error}
--------------------------------------------------------------------------------
error at `(...1)`: i32 can't be spread

================================================================================
Tuples with named fields can be mut
================================================================================
let m = mut (x: 1, y: [1]);
m.x = 2;
--------------------------------------------------------------------------------
let m: mut (x: i32, y: imm [i32])
m.x = 2: ()

================================================================================
A field of a tuple that isn't mut can't be assigned
:error
================================================================================
let m = (x: 1);
m.x = 2;
--------------------------------------------------------------------------------
let m: imm (x: i32)
m.x = 2: ()
--------------------------------------------------------------------------------
error at `m`: can't change imm (x: i32)

================================================================================
Patterns with named fields
================================================================================
let p = (1, x: 2, y: True);
let a = match p { (0, x: n, y: _) => n, (_, x: n, ..) => n };
let (k, ..) = p;
let (y: flag, ..) = p;
--------------------------------------------------------------------------------
let p: imm (i32, x: i32, y: Bool)
let a: i32
let (k, ..): imm (i32, x: i32, y: Bool)
let (y: flag, ..): imm (i32, x: i32, y: Bool)

================================================================================
A pattern without .. names every field
:error
================================================================================
let p = (x: 1, y: 2);
let (x: a) = p;
--------------------------------------------------------------------------------
let p: imm (x: i32, y: i32)
let (x: a): imm (x: i32, y: i32)
--------------------------------------------------------------------------------
error at `(x: a)`: expected imm (x: {integer}, y: {integer}), found (x: a)

================================================================================
A pattern names fields the type has
:error
================================================================================
let p = (x: 1, y: 2);
let (z: a, ..) = p;
--------------------------------------------------------------------------------
let p: imm (x: i32, y: i32)
let (z: a, ..): imm (x: i32, y: i32)
--------------------------------------------------------------------------------
error at `(z: a, ..)`: imm (x: i32, y: i32) has no field z

================================================================================
Rests and pack parameters aren't supported yet
================================================================================
fn g[*Ts](t: (i32, *Ts)) = t;
--------------------------------------------------------------------------------
g: fn({error}) -> {error}
--------------------------------------------------------------------------------
warning at `(i32, *Ts)`: unsupported
