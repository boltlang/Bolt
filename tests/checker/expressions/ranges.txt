================================================================================
A Range is looped over with for
:prelude
================================================================================
let r = Range { start: 0, end: 3 };
let mut sum = 0;
for i in r { sum += i; }
for i in r { sum += i; }
let bytes = Range { start: 1u8, end: 4u8 };
for b in bytes { b; }
--------------------------------------------------------------------------------
let r: imm Range[i32]
let sum: i32
for i in r { sum += i; }: ()
for i in r { sum += i; }: ()
let bytes: imm Range[u8]
for b in bytes { b; }: ()

================================================================================
A Range needs values that come one after another
:error
:prelude
================================================================================
let words = Range { start: "a", end: "c" };
for s in words { s; }
--------------------------------------------------------------------------------
let words: imm Range[String]
for s in words { s; }: ()
--------------------------------------------------------------------------------
error at `words`: no instance Step[String]
error at `for s in words { s; }`: no instance Step[String]

================================================================================
a..b is a Range
:prelude
================================================================================
let r = 0..3;
let bytes = 1u8..4;
let mut sum = 0;
for i in 0..10 { sum += i; }
fn count(n: i32) -> i32 { let mut c = 0; for _ in 0..n { c += 1; } c }
--------------------------------------------------------------------------------
count: fn(n: i32) -> i32
let r: imm Range[i32]
let bytes: imm Range[u8]
let sum: i32
for i in 0..10 { sum += i; }: ()

================================================================================

:error

================================================================================

================================================================================
a..=b is a RangeInclusive
:prelude
================================================================================

let all = 0u8..=255;
for i in 1..=10 { i; }
--------------------------------------------------------------------------------
let r: imm RangeInclusive[i32]
let all: imm RangeInclusive[u8]
for i in 1..=10 { i; }: ()

--------------------------------------------------------------------------------
