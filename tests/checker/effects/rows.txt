================================================================================
A written row allows what's in it
:test-prelude
================================================================================
effect Log { fn log(message: String); }
fn greet(name: String) -> () does Log { log(name); }
--------------------------------------------------------------------------------
greet: fn(String) -> () does Log

================================================================================
A written row allows nothing else
:test-prelude
:error
================================================================================
effect Log { fn log(message: String); }
effect Async { fn wait(); }
fn greet(name: String) does Log { log(name); wait(); }
--------------------------------------------------------------------------------
greet: fn(String) -> () does Log
--------------------------------------------------------------------------------
error at `wait()`: does Async, only Log allowed

================================================================================
A result type without does means no effects
:test-prelude
:error
================================================================================
effect Log { fn log(message: String); }
fn quiet() -> () { log("hi"); }
--------------------------------------------------------------------------------
quiet: fn() -> ()
--------------------------------------------------------------------------------
error at `log("hi")`: does Log, only Nothing allowed

================================================================================
Nothing is the row without effects
:test-prelude
:error
================================================================================
effect Log { fn log(message: String); }
fn double(x: i32) does Nothing = x * 2;
fn noisy() does Nothing { log("hi"); }
--------------------------------------------------------------------------------
double: fn(i32) -> i32
noisy: fn() -> ()
--------------------------------------------------------------------------------
error at `log("hi")`: does Log, only Nothing allowed

================================================================================
A closure's effects are its own
:test-prelude
================================================================================
effect Log { fn log(message: String); }
fn make() = || log("hi");
fn run() { let f = || log("hi"); f(); }
--------------------------------------------------------------------------------
make: fn() -> fn() -> () does Log
run: fn() -> () does Log

================================================================================
A function takes the effects of the function it's given
:test-prelude
================================================================================
effect Log { fn log(message: String); }
fn map[A, B, E](x: A, f: fn(A) -> B does E) -> B does E = f(x);
fn noisy(x: i32) = map(x, |y| { log("y"); y });
fn quiet(x: i32) = map(x, |y| y + 1);
--------------------------------------------------------------------------------
map: forall A B E. fn(A, fn(A) -> B does E) -> B does E
noisy: fn(i32) -> i32 does Log
quiet: fn(i32) -> i32

================================================================================
A function without effects takes no function with effects
:test-prelude
:error
================================================================================
effect Log { fn log(message: String); }
fn run(f: fn() -> ()) -> () = f();
fn g() { run(|| log("hi")); }
--------------------------------------------------------------------------------
run: fn(fn() -> ()) -> ()
g: fn() -> ()
--------------------------------------------------------------------------------
error at `log("hi")`: does Log, only Nothing allowed

================================================================================
An effect in a row has the same arguments everywhere

--------------------------------------------------------------------------------
