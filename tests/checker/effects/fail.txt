================================================================================
or_fail and attempt go between Result and Fail
:prelude
================================================================================
fn parse(s: String) -> Result[String, i32] = if s == "42" { Ok(42) } else { Err("not a number") };
fn twice(s: String) -> i32 does Fail[String] = parse(s).or_fail() * 2;
fn both() = attempt(|| twice("42") + twice("x"));
fn direct() = attempt(|| fail("no"));
--------------------------------------------------------------------------------
parse: fn(String) -> Result[String, i32]
twice: fn(String) -> i32 does Fail[String]
both: fn() -> Result[String, i32]
direct: forall a. fn() -> Result[String, a]

================================================================================
attempt leaves the other effects of what it runs
:prelude
================================================================================
effect Log { fn log(m: String); }
fn step() -> i32 does Log + Fail[String] { log("step"); fail("stop") }
fn written() -> Result[String, i32] does Log = attempt(step);
fn inferred() = attempt(|| { log("a"); step() });
--------------------------------------------------------------------------------
step: fn() -> i32 does Log + Fail[String]
written: fn() -> Result[String, i32] does Log
inferred: fn() -> Result[String, i32] does Log

================================================================================
fn leaks() -> i32 = Err("no").or_fail();

leaks: fn() -> i32
