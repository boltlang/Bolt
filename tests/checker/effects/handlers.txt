================================================================================
A handler takes its effect out of the row
:test-prelude
================================================================================
effect Yield { fn yield(value: i32); }
fn numbers() -> () does Yield { yield(1); yield(2); }
fn sum() -> i32 = match numbers() {
  () => 0,
  effect yield(x), k => x + resume(k, ()),
};
--------------------------------------------------------------------------------
numbers: fn() -> () does Yield
sum: fn() -> i32

================================================================================
Resuming a continuation can run the arms again
:error
:test-prelude
================================================================================
effect Yield { fn yield(value: i32); }
effect Log { fn log(message: String); }
fn later(k: Cont[(), (), Nothing]) { resume(k, ()) }
fn later_with[E](k: Cont[(), (), E]) does E { resume(k, ()) }

  () => (),
  effect yield(x), k => { log("yielded"); later(k) },
};

  () => (),
  effect yield(x), k => { log("yielded"); later_with(k) },
};
--------------------------------------------------------------------------------

================================================================================
A closure given where an open row is expected keeps its other effects
:test-prelude
================================================================================
effect Log { fn log(m: String); }
effect Boom { fn boom[T]() -> T; }
fn catch_boom[T, R](body: fn() -> T does Boom + R) -> Option[T] does R = match body() {
  v => Some(v),
  effect boom(), _ => None,
};
fn written() -> Option[i32] does Log = catch_boom(|| { log("a"); boom() });
fn inferred() = catch_boom(|| { log("a"); boom() });
fn pure() = catch_boom(|| boom());
--------------------------------------------------------------------------------
catch_boom: forall T R. fn(body: fn() -> T does Boom + R) -> Option[T] does R
written: fn() -> Option[i32] does Log
inferred: forall a. fn() -> Option[a] does Log
pure: forall a. fn() -> Option[a]
