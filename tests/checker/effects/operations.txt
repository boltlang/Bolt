================================================================================
An operation is called like a function
:test-prelude
================================================================================
effect Log { fn log(message: String); }
fn greet(name: String) { log(name); }
greet("Bolt");
--------------------------------------------------------------------------------
greet: fn(String) -> ()
greet("Bolt"): ()

================================================================================
An operation checks its arguments
:test-prelude
:error
================================================================================
effect Log { fn log(message: String); }
log(1);
--------------------------------------------------------------------------------
log(1): ()
--------------------------------------------------------------------------------
error at `1`: expected String, found {integer}

================================================================================
An operation of an effect with parameters
:test-prelude
================================================================================
effect State[S] { fn get() -> S; fn set(value: S); }
fn bump() { set(get() + 1); }
--------------------------------------------------------------------------------
bump: fn() -> ()

================================================================================
An operation through its effect
:test-prelude
================================================================================

--------------------------------------------------------------------------------

================================================================================
An operation with type parameters of its own
:test-prelude
================================================================================
effect Raise[E] { fn raise[T](error: E) -> T; }
fn divide(a: i32, b: i32) -> i32 = if b == 0 { raise("division by zero") } else { a / b };
--------------------------------------------------------------------------------
divide: fn(i32, i32) -> i32
