struct CountIter { left: [i32] }
impl Iterator[mut CountIter] {
  type Item = i32;
  fn next(it) = if it.left[0] > 0 { it.left[0] -= 1; Some(it.left[0]) } else { None };
}
impl Iterable[Count] {
  type Iter = mut CountIter;
  fn iter(c) = mut CountIter { left: mut [c.n] };
}
fn f(c: Count) {
  for x in c {
    let y: i32 = x;

}

Generic functions over the classes
================================================================================
fn first[C](c: C) -> C.Item where Index[C] = c[0];
fn count[C](c: C) -> i32 where Iterable[C], Iterator[C.Iter] {
  let mut n = 0;
  for x in c { n += 1; }
  n
}
