================================================================================

let (c, .., d) = (1, 2, "x");

================================================================================
If let and let else
================================================================================

fn get(o, default) = if let Option.Some(x) = o { x } else { default };
fn unwrap(o: Option[i32]) -> i32 {
  let Option.Some(x) = o else { return 0; };
  x
}

--------------------------------------------------------------------------------
