================================================================================
Data types and patterns
================================================================================
enum Option[T] { None, Some(T) }
struct Point { x: i32, y: i32 }
fn get(o, d) = match o { Option.Some(v) => v, Option.None => d };
fn norm(p) = p.x * p.x;
let p = Point { x: 1, y: 2 };
let Point { x, .. } = p;
get(Option.Some(p), Point { x: 0 });
p.z;
--------------------------------------------------------------------------------
get: forall a. fn(Option[a], a) -> a
norm: forall a b. a.x: b => fn(a) -> b
let p: Point
let Point { x, .. }: Point
get(Option.Some(p), Point { x: 0 }): Point
p.z: {error}
--------------------------------------------------------------------------------
error at `Point { x: 0 }`: missing y
error at `p.z`: Point has no field z

================================================================================
Fields of unknown types become part of the function type
================================================================================
struct Point { x: i32, y: i32 }
struct Size { x: u8 }
struct Line { start: Point }
fn get_x(p) = p.x;
fn start_x(l) = l.start.x;
get_x(Point { x: 1, y: 2 });
get_x(Size { x: 3u8 });
start_x(Line { start: Point { x: 1, y: 2 } });
get_x((1, 2));
let t = (true, 1);
t.0;
--------------------------------------------------------------------------------
get_x: forall a b. a.x: b => fn(a) -> b
start_x: forall a b c. a.start: c => c.x: b => fn(a) -> b
get_x(Point { x: 1, y: 2 }): i32
get_x(Size { x: 3u8 }): u8
start_x(Line { start: Point { x: 1, y: 2 } }): i32
get_x((1, 2)): {error}
let t: (bool, i32)
t.0: bool
--------------------------------------------------------------------------------
error at `get_x`: (i32, i32) has no field x

Generic structs
================================================================================
struct Pair[A, B] { first: A, second: B }
fn swap(p) = Pair { first: p.second, second: p.first };
let p = Pair { first: 1, second: "a" };
p.first;
let q: Pair[bool, i32] = Pair { first: 1, second: true };

--------------------------------------------------------------------------------
swap: forall a b c. a.second: b => a.first: c => fn(a) -> Pair[b, c]
let p: Pair[i32, str]
p.first: i32
let q: Pair[bool, i32]
--------------------------------------------------------------------------------
error at `Pair { first: 1, second: true }`: expected Pair[bool, i32], found Pair[{integer}, bool]

================================================================================
Variants used as functions
================================================================================
enum Option[T] { Some(T), None }
fn map(f, o) = match o { Option.Some(x) => Option.Some(f(x)), Option.None => Option.None };
let wrap = Option.Some;
wrap(1);
--------------------------------------------------------------------------------
map: forall a b. fn(fn(a) -> b, Option[a]) -> Option[b]
let wrap: fn(i32) -> Option[i32]
wrap(1): Option[i32]
