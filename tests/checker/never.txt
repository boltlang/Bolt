================================================================================
Panic has any type
================================================================================
extern "bolt" { fn panic(message: String) -> !; }
fn f(x: i32) -> String = if x > 0 { "positive" } else { panic("negative") };
fn g() -> i32 = panic("never");
let a: Bool = panic("no");

--------------------------------------------------------------------------------
f: fn(i32) -> String
g: fn() -> i32
let a: Bool

================================================================================
A function that never returns
================================================================================
extern "bolt" { fn panic(message: String) -> !; }

fn f(x: i32) -> i32 = if x > 0 { x } else { fail("negative") };

--------------------------------------------------------------------------------
fail: fn(String) -> !

================================================================================

================================================================================
Matching an empty enum
================================================================================
