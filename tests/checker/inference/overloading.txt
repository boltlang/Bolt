================================================================================
Functions with the same name are chosen by their argument types
:test-prelude
================================================================================
struct Point { x: i32, y: i32 }
fn size(x: i32) -> i32 = x;
fn size(p: Point) -> i32 = p.x + p.y;
fn size(b: Bool) -> i32 = if b { 1 } else { 0 };
size(3);
size(Point { x: 1, y: 2 });
size(True);
--------------------------------------------------------------------------------
size: fn(x: i32) -> i32
size: fn(p: Point) -> i32
size: fn(b: Bool) -> i32
size(3): i32
size(Point { x: 1, y: 2 }): i32
size(True): i32

================================================================================

:test-prelude

fn size(x: i32) -> i32 = x;
