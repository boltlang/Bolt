================================================================================
A function applied to itself has an infinite type
:error
================================================================================
fn f(x) = x(x);
--------------------------------------------------------------------------------
f: forall a b e. fn(x: fn(a) -> b does e) -> b does e
--------------------------------------------------------------------------------
error at `x`: infinite type fn(a) -> b

================================================================================
A function that returns itself has an infinite type
:error
================================================================================
fn f(x) = (x, f);
--------------------------------------------------------------------------------
f: forall a b. a: imm => fn(x: a) -> b
--------------------------------------------------------------------------------
error at `(x, f)`: infinite type imm (a, fn(x: a) -> b)

================================================================================
A list that contains itself has an infinite type
:error
================================================================================
fn f(x) = [x, [x]];
--------------------------------------------------------------------------------
f: forall a. a: imm => fn(x: a) -> imm [a]
--------------------------------------------------------------------------------
error at `[x]`: infinite type imm [a]

================================================================================
Polymorphic recursion with a signature
================================================================================
fn f[T](x: T) -> i32 = { f(1); f(True); 0 };
--------------------------------------------------------------------------------
f: forall T. fn(x: T) -> i32

================================================================================
Polymorphic recursion through a nested type
:test-prelude
================================================================================
enum Nested[T] { Flat(T), Nest(Nested[[T]]) }
fn depth[T](n: Nested[T]) -> i32 = match n {
  Nested.Flat(_) => 0,
  Nested.Nest(m) => 1 + depth(m),
};
--------------------------------------------------------------------------------
depth: forall T. fn(n: Nested[T]) -> i32

================================================================================
Polymorphic recursion without a signature
:error
:test-prelude
================================================================================
enum Nested[T] { Flat(T), Nest(Nested[[T]]) }
fn depth(n) = match n {
  Nested.Flat(_) => 0,
  Nested.Nest(m) => 1 + depth(m),
};
--------------------------------------------------------------------------------
depth: forall a. fn(n: Nested[a]) -> i32
--------------------------------------------------------------------------------
error at `m`: infinite type imm [a]

================================================================================
Mutual recursion where one function has a signature
================================================================================
fn a[T](x: T) -> T = b(x);
fn b(x) = a(x);
a(1);
b(True);
--------------------------------------------------------------------------------
a: forall T. fn(x: T) -> T
b: forall a. fn(x: a) -> a
a(1): i32
b(True): Bool

================================================================================
Mutual recursion without anything known
================================================================================
fn f(x) = g(x);
fn g(x) = f(x);
--------------------------------------------------------------------------------
f: forall a b. fn(x: a) -> b
g: forall a b. fn(x: a) -> b

================================================================================
Mutually recursive functions are generic only outside their group
================================================================================
fn f(x) = { g(1); x };
fn g(y) = { f(True); y };
--------------------------------------------------------------------------------
f: fn(x: Bool) -> Bool
g: fn(y: i32) -> i32

================================================================================
A group of three
:test-prelude
================================================================================
fn a(n) = if n == 0 { 0 } else { b(n - 1) };
fn b(n) = c(n);
fn c(n) = a(n) + 1;
--------------------------------------------------------------------------------
a: fn(n: i32) -> i32
b: fn(n: i32) -> i32
c: fn(n: i32) -> i32

================================================================================
A lambda can't call itself through its let
:error
================================================================================

--------------------------------------------------------------------------------
f: fn() -> ()
--------------------------------------------------------------------------------
error at `g`: unresolved
