================================================================================
A literal in a function without a signature defaults when generalized
:test-prelude
================================================================================
fn inc(x) = x + 1;
inc(1u8);
--------------------------------------------------------------------------------
inc: fn(x: i32) -> i32
inc(1u8): i32
--------------------------------------------------------------------------------
error at `1u8`: expected i32, found u8

================================================================================
A function that returns a literal can't be used at another type
================================================================================
fn zero() = 0;
let a: u8 = zero();
--------------------------------------------------------------------------------
zero: fn() -> i32
let a: u8
--------------------------------------------------------------------------------
error at `zero()`: expected u8, found i32

================================================================================
An empty array from a function is generic
================================================================================
fn empty() = [];
let a: [i32] = empty();
let b: [Bool] = empty();
--------------------------------------------------------------------------------
empty: forall a. a: imm => fn() -> imm [a]
let a: imm [i32]
let b: imm [Bool]

================================================================================
A None from a function is generic

let b: Option[String] = none();
--------------------------------------------------------------------------------
none: forall a. fn() -> Option[a]

================================================================================
Top-level lets are not generic
================================================================================

--------------------------------------------------------------------------------

id(1): i32
id(True): i32
--------------------------------------------------------------------------------
error at `True`: expected {integer}, found Bool

================================================================================
Class constraints in lambdas are inferred for their function
:test-prelude
================================================================================
class Display[T] { fn display(x: T) -> String; }
fn f(x) = (|y| display(y))(x);
fn g(x) { let s = |y| display(y); s(x) }
--------------------------------------------------------------------------------
f: forall a. Display[a] => fn(x: a) -> String
g: forall a. Display[a] => fn(x: a) -> String

================================================================================
Generic functions used at several types in one function
:test-prelude
================================================================================
class Display[T] { fn display(x: T) -> String; }
impl Display[i32] { fn display(x) = "int"; }
impl Display[Bool] { fn display(x) = "Bool"; }
fn both() = (display(1), display(True));
fn twice(x) = (display(x), display(x));
fn mixed(x) = (twice(x), twice(1));
--------------------------------------------------------------------------------
Display[i32].display: fn(x: i32) -> String
Display[Bool].display: fn(x: Bool) -> String
both: fn() -> imm (String, String)
twice: forall a. Display[a] => fn(x: a) -> imm (String, String)
mixed: forall a. Display[a] => fn(x: a) -> imm (imm (String, String), imm (String, String))
