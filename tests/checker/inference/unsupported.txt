================================================================================
Unsupported expressions have the error type
================================================================================
fn f(x) { x?; }
fn h() = 1..;
--------------------------------------------------------------------------------
f: forall a. fn(a) -> ()
h: fn() -> {error}
--------------------------------------------------------------------------------
warning at `x?`: unsupported
warning at `1..`: unsupported

================================================================================
Unsupported types have the error type
================================================================================
fn f(x: impl Show) = x;
--------------------------------------------------------------------------------
f: fn({error}) -> {error}
--------------------------------------------------------------------------------
warning at `impl Show`: unsupported

================================================================================
Arithmetic needs an impl of its class
:test-prelude
================================================================================
"a" + "b";
True * False;
-"a";
let x = 1u8;
-x;
--------------------------------------------------------------------------------
"a" + "b": {error}
True * False: {error}
-"a": {error}
let x: u8
-x: {error}
--------------------------------------------------------------------------------
error at `"a" + "b"`: no instance Add[String]
error at `True * False`: no instance Mul[Bool]
error at `-"a"`: no instance Neg[String]
error at `-x`: no instance Neg[u8]
