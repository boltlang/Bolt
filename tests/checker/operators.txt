================================================================================
Arithmetic keeps the type of its operands
================================================================================
1u8 + 2;
3.0 / 2.0;
7 % 2;
let x = 1i64;
x * x - x;

--------------------------------------------------------------------------------
1u8 + 2: u8
3.0 / 2.0: f64
7 % 2: i32
let x: i64
x * x - x: i64

================================================================================
Arithmetic needs operands of the same type
================================================================================
1u8 + 1i32;
1 + True;
"a" - 1;

--------------------------------------------------------------------------------
1u8 + 1i32: u8
1 + True: i32
"a" - 1: String
--------------------------------------------------------------------------------
error at `1i32`: expected u8, found i32
error at `True`: expected {integer}, found Bool
error at `1`: expected String, found {integer}

================================================================================
Shifts can shift by any integer type

--------------------------------------------------------------------------------

================================================================================

--------------------------------------------------------------------------------

================================================================================
Logical operators take and give bools

--------------------------------------------------------------------------------

================================================================================
Negation keeps the type
================================================================================
-1;
-1.5f32;
let x = 2u8;
-x;

--------------------------------------------------------------------------------
-1: i32
-1.5f32: f32
let x: u8

================================================================================
Not of an unknown type is a Bool
================================================================================
fn f(x) = !x;

--------------------------------------------------------------------------------
f: fn(Bool) -> Bool

================================================================================
Assignments give unit
================================================================================
fn f() {
  let mut x = 1;
  x = 2;
  x += 3;
  x = True;
}

--------------------------------------------------------------------------------
f: fn() -> ()
--------------------------------------------------------------------------------
error at `True`: expected {integer}, found Bool

================================================================================
Assigning to a variable that isn't mut is an error
================================================================================
struct Point { x: i32 }
fn f(p: Point, xs: [i32; 2]) {
  let x = 1;
  x = 2;
  x += 3;
  p.x = 1;
  xs[0] = 1;
  let mut y = 1;
  y = 2;
  let mut q = p;
  q.x = 2;
  let z;
  z = 1;
  z = 2;
}

--------------------------------------------------------------------------------
f: fn(Point, [i32]) -> ()
--------------------------------------------------------------------------------
error at `x`: not mut
error at `x`: not mut
error at `p.x`: not mut
error at `xs[0]`: not mut
error at `z`: not mut

================================================================================
Assigning to something that isn't a place is an error
================================================================================
struct Point { x: i32 }
const C: i32 = 1;
fn g() -> i32 = 1;
fn h() -> Point = Point { x: 1 };
fn f() {
  1 = 2;
  g() = 3;
  g = g;
  C += 1;
  h().x = 2;
  (1, 2).0 = 3;
  nope = 1;
}

--------------------------------------------------------------------------------
g: fn() -> i32
h: fn() -> Point
f: fn() -> ()
--------------------------------------------------------------------------------
error at `1`: not a place
error at `g()`: not a place
error at `g`: not a place
error at `C`: not a place
error at `h().x`: not a place
error at `(1, 2).0`: not a place
error at `nope`: unresolved
