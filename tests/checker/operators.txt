================================================================================
Arithmetic keeps the type of its operands
================================================================================
1u8 + 2;
3.0 / 2.0;
7 % 2;
let x = 1i64;
x * x - x;

--------------------------------------------------------------------------------
1u8 + 2: u8
3.0 / 2.0: f64
7 % 2: i32
let x: i64
x * x - x: i64

================================================================================
Arithmetic needs operands of the same type
================================================================================
1u8 + 1i32;
1 + true;
"a" - 1;

--------------------------------------------------------------------------------
1u8 + 1i32: u8
1 + true: i32
"a" - 1: str
--------------------------------------------------------------------------------
error at `1i32`: expected u8, found i32
error at `true`: expected {integer}, found bool
error at `1`: expected str, found {integer}

================================================================================
Shifts can shift by any integer type
================================================================================
1u8 << 3u32;
1i64 >> 2;

--------------------------------------------------------------------------------
1u8 << 3u32: u8
1i64 >> 2: i64

================================================================================
Comparisons give a bool
================================================================================
1 == 2;
1.5 < 2.0;
"a" != "b";
1 <= true;

--------------------------------------------------------------------------------
1 == 2: bool
1.5 < 2.0: bool
"a" != "b": bool
1 <= true: bool
--------------------------------------------------------------------------------
error at `true`: expected {integer}, found bool

================================================================================
Logical operators take and give bools
================================================================================
true && false;
!true || false;
1 && true;

--------------------------------------------------------------------------------
true && false: bool
!true || false: bool
1 && true: bool
--------------------------------------------------------------------------------
error at `1`: expected bool, found {integer}

================================================================================
Negation keeps the type
================================================================================
-1;
-1.5f32;
let x = 2u8;
-x;

--------------------------------------------------------------------------------
-1: i32
-1.5f32: f32
let x: u8
-x: u8

================================================================================
Not of an unknown type is a bool
================================================================================
fn f(x) = !x;

--------------------------------------------------------------------------------
f: fn(bool) -> bool

================================================================================
Assignments give unit
================================================================================
fn f() {
  let mut x = 1;
  x = 2;
  x += 3;
  x = true;
}

--------------------------------------------------------------------------------
f: fn() -> ()
--------------------------------------------------------------------------------
error at `true`: expected {integer}, found bool

================================================================================
Custom operators call their definition
================================================================================
fn[T] (a: T) <> (b: T) -> (T, T) = (a, b);
1 <> 2;
true <> false;
1 <> true;

--------------------------------------------------------------------------------
<>: forall T. fn(T, T) -> (T, T)
1 <> 2: (i32, i32)
true <> false: (bool, bool)
1 <> true: (i32, i32)
--------------------------------------------------------------------------------
error at `true`: expected {integer}, found bool

================================================================================
A custom operator that isn't defined
================================================================================
1 <$> 2;

--------------------------------------------------------------------------------
1 <$> 2: {error}
--------------------------------------------------------------------------------
error at `1 <$> 2`: unresolved

================================================================================
Precedence of mixed operators
================================================================================
1 + 2 * 3 == 7 && !false;
1 < 2 == true;

--------------------------------------------------------------------------------
1 + 2 * 3 == 7 && !false: bool
1 < 2 == true: bool

================================================================================
Bitwise operators
================================================================================
1 & 2;
1u8 | 2;
true ^ false;
1 & true;

--------------------------------------------------------------------------------
1 & 2: i32
1u8 | 2: u8
true ^ false: bool
1 & true: i32
--------------------------------------------------------------------------------
error at `true`: expected {integer}, found bool

Remainder of floats
================================================================================

================================================================================
Negation of an unknown type
================================================================================
fn neg(x) = -x;
neg(1);
neg(1.5);

--------------------------------------------------------------------------------

neg(1): i32
neg(1.5): f64

================================================================================
Comparing functions and tuples
================================================================================
fn inc(x: i32) -> i32 = x + 1;
inc == inc;
(1, true) == (1, true);
(1, true) < (2, false);

--------------------------------------------------------------------------------
inc: fn(i32) -> i32
inc == inc: bool
(1, true) == (1, true): bool
(1, true) < (2, false): bool

================================================================================
Operators on the results of calls
================================================================================
fn one() -> i32 = 1;
fn yes() -> bool = true;
one() + one();
one() + yes();

--------------------------------------------------------------------------------
one: fn() -> i32
yes: fn() -> bool
one() + one(): i32
one() + yes(): i32
--------------------------------------------------------------------------------
error at `yes()`: expected i32, found bool

================================================================================
Custom operators with a signature that doesn't fit
:error
================================================================================
fn (a: i32) <+> (b: i32) -> i32 = a + b;
1 <+> true;
true <+> 1;

--------------------------------------------------------------------------------
<+>: fn(i32, i32) -> i32
1 <+> true: i32
true <+> 1: i32
--------------------------------------------------------------------------------
error at `true`: expected i32, found bool
error at `true`: expected i32, found bool

================================================================================
Custom operators without types
================================================================================
fn a <&> b = (b, a);
1 <&> true;
--------------------------------------------------------------------------------
<&>: forall a b. fn(a, b) -> (b, a)
1 <&> true: (bool, i32)
