================================================================================
Arithmetic keeps the type of its operands
================================================================================
1u8 + 2;
3.0 / 2.0;
7 % 2;
let x = 1i64;
x * x - x;

--------------------------------------------------------------------------------
1u8 + 2: u8
3.0 / 2.0: f64
7 % 2: i32
let x: i64
x * x - x: i64

================================================================================
Arithmetic needs operands of the same type
================================================================================
1u8 + 1i32;
1 + true;
"a" - 1;

--------------------------------------------------------------------------------
1u8 + 1i32: u8
1 + true: i32
"a" - 1: str
--------------------------------------------------------------------------------
error at `1i32`: expected u8, found i32
error at `true`: expected {integer}, found bool
error at `1`: expected str, found {integer}

================================================================================
Shifts can shift by any integer type
================================================================================
1u8 << 3u32;
1i64 >> 2;

--------------------------------------------------------------------------------
1u8 << 3u32: u8
1i64 >> 2: i64

================================================================================
Comparisons give a bool
================================================================================
1 == 2;
1.5 < 2.0;
"a" != "b";
1 <= true;

--------------------------------------------------------------------------------
1 == 2: bool
1.5 < 2.0: bool
"a" != "b": bool
1 <= true: bool
--------------------------------------------------------------------------------
error at `true`: expected {integer}, found bool

================================================================================
Logical operators take and give bools
================================================================================
true && false;
!true || false;
1 && true;

--------------------------------------------------------------------------------
true && false: bool
!true || false: bool
1 && true: bool
--------------------------------------------------------------------------------
error at `1`: expected bool, found {integer}

================================================================================
Negation keeps the type
================================================================================
-1;
-1.5f32;
let x = 2u8;
-x;

--------------------------------------------------------------------------------
-1: i32
-1.5f32: f32
let x: u8
-x: u8

================================================================================
Not of an unknown type is a bool
================================================================================
fn f(x) = !x;

--------------------------------------------------------------------------------
f: fn(bool) -> bool

================================================================================
Assignments give unit
================================================================================
fn f() {
  let mut x = 1;
  x = 2;
  x += 3;
  x = true;
}

--------------------------------------------------------------------------------
f: fn() -> ()
--------------------------------------------------------------------------------
error at `true`: expected {integer}, found bool

================================================================================
Assigning to a variable that isn't mut is an error
================================================================================
struct Point { x: i32 }
fn f(p: Point, xs: [i32; 2]) {
  let x = 1;
  x = 2;
  x += 3;
  p.x = 1;
  xs[0] = 1;
  let mut y = 1;
  y = 2;
  let mut q = p;
  q.x = 2;
  let z;
  z = 1;
  z = 2;
}

--------------------------------------------------------------------------------
f: fn(Point, [i32]) -> ()
--------------------------------------------------------------------------------
error at `x`: not mut
error at `x`: not mut
error at `p.x`: not mut
error at `xs[0]`: not mut
error at `z`: not mut

================================================================================
Assigning to something that isn't a place is an error
================================================================================
struct Point { x: i32 }
const C: i32 = 1;
fn g() -> i32 = 1;
fn h() -> Point = Point { x: 1 };
fn f() {
  1 = 2;
  g() = 3;
  g = g;
  C += 1;
  h().x = 2;
  (1, 2).0 = 3;
  nope = 1;
}

--------------------------------------------------------------------------------
g: fn() -> i32
h: fn() -> Point
f: fn() -> ()
--------------------------------------------------------------------------------
error at `1`: not a place
error at `g()`: not a place
error at `g`: not a place
error at `C`: not a place
error at `h().x`: not a place
error at `(1, 2).0`: not a place
error at `nope`: unresolved

================================================================================
Custom operators call their definition
================================================================================
fn[T] (a: T) <> (b: T) -> (T, T) = (a, b);
1 <> 2;
true <> false;
1 <> true;

--------------------------------------------------------------------------------
<>: forall T. fn(T, T) -> (T, T)
1 <> 2: (i32, i32)
true <> false: (bool, bool)
1 <> true: (i32, i32)
--------------------------------------------------------------------------------
error at `true`: expected {integer}, found bool

================================================================================
A custom operator that isn't defined
================================================================================
1 <$> 2;

--------------------------------------------------------------------------------
1 <$> 2: {error}
--------------------------------------------------------------------------------
error at `1 <$> 2`: unresolved
