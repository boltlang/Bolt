================================================================================
A function applied to itself has an infinite type
:error
================================================================================
fn f(x) = x(x);

--------------------------------------------------------------------------------
f: forall a b. fn(fn(a) -> b) -> b
--------------------------------------------------------------------------------
error at `x`: infinite type fn(a) -> b

================================================================================
A function that returns itself has an infinite type
:error
================================================================================
fn f(x) = (x, f);

--------------------------------------------------------------------------------
f: forall a b. fn(a) -> b
--------------------------------------------------------------------------------
error at `(x, f)`: infinite type (a, fn(a) -> b)

================================================================================
A list that contains itself has an infinite type
:error
================================================================================
fn f(x) = [x, [x]];

--------------------------------------------------------------------------------
f: forall a. fn(a) -> [a]
--------------------------------------------------------------------------------
error at `[x]`: infinite type [a]

================================================================================
Polymorphic recursion with a signature
================================================================================
fn f[T](x: T) -> i32 = { f(1); f(True); 0 };

--------------------------------------------------------------------------------
f: forall T. fn(T) -> i32

================================================================================
Polymorphic recursion through a nested type
================================================================================
enum Nested[T] { Flat(T), Nest(Nested[[T]]) }
fn depth[T](n: Nested[T]) -> i32 = match n {
  Nested.Flat(_) => 0,
  Nested.Nest(m) => 1 + depth(m),
};

--------------------------------------------------------------------------------
depth: forall T. fn(Nested[T]) -> i32

================================================================================
Polymorphic recursion without a signature
:error
================================================================================
enum Nested[T] { Flat(T), Nest(Nested[[T]]) }
fn depth(n) = match n {
  Nested.Flat(_) => 0,
  Nested.Nest(m) => 1 + depth(m),
};

--------------------------------------------------------------------------------
depth: forall a. fn(Nested[a]) -> i32
--------------------------------------------------------------------------------
error at `m`: infinite type [a]

================================================================================
Mutual recursion where one function has a signature
================================================================================
fn a[T](x: T) -> T = b(x);
fn b(x) = a(x);
a(1);
b(True);

--------------------------------------------------------------------------------
a: forall T. fn(T) -> T
b: forall a. fn(a) -> a
a(1): i32
b(True): Bool

================================================================================
Mutual recursion without anything known
================================================================================
fn f(x) = g(x);
fn g(x) = f(x);

--------------------------------------------------------------------------------
f: forall a b. fn(a) -> b
g: forall a b. fn(a) -> b

================================================================================
Mutually recursive functions are generic only outside their group
================================================================================
fn f(x) = { g(1); x };
fn g(y) = { f(True); y };

--------------------------------------------------------------------------------
f: fn(Bool) -> Bool
g: fn(i32) -> i32

================================================================================
A group of three
================================================================================
fn a(n) = if n == 0 { 0 } else { b(n - 1) };
fn b(n) = c(n);
fn c(n) = a(n) + 1;

--------------------------------------------------------------------------------
a: fn(i32) -> i32
b: fn(i32) -> i32
c: fn(i32) -> i32

================================================================================
A lambda can't call itself through its let
:error
================================================================================
fn f() { let g = |x| g(x); }

--------------------------------------------------------------------------------
f: fn() -> ()
--------------------------------------------------------------------------------
error at `g`: unresolved

================================================================================
A let can't refer to itself
:error
================================================================================
fn f() { let x = x + 1; }
--------------------------------------------------------------------------------
f: fn() -> ()
--------------------------------------------------------------------------------
error at `x`: unresolved
