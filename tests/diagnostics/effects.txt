================================================================================
An effect isn't a type
:test-prelude
================================================================================
effect Log { fn log(message: String); }
effect State[S] { fn get() -> S; }
fn f(x: Log) {}
      //^^^ error: expected a type of kind `Type`, found `Log` of kind `Effect`
fn g(x: State[i32]) {}
      //^^^^^^^^^^ error: expected a type of kind `Type`, found `State[i32]` of kind `Effect`
fn h(x: State) {}
      //^^^^^ error: expected a type of kind `Type`, found `State` of kind `Type -> Effect`

================================================================================
Effects that the code around can't do
:test-prelude
================================================================================
effect Log { fn log(message: String); }
effect Async { fn wait(); }
fn quiet() -> () { log("hi"); }
                 //^^^^^^^^^ error: this does `Log`, but the code it's in has no effects
fn greet() -> () does Log { wait(); }
                          //^^^^^^ error: this does `Async`, but the code it's in only does `Log`

================================================================================
An effect arm handles an operation of an effect
:test-prelude
================================================================================
effect State { fn get() -> i32; }
effect Config { fn get(key: String) -> String; }
fn helper() -> i32 = 0;
fn run(body: fn() -> i32 does State + Config) -> i32 = match body() {
  v => v,
  effect helper(), k => 0,
       //^^^^^^^^ error: this isn't an operation of an effect, so an effect arm can't handle it
  effect get(), k => resume(k, 0),
       //^^^^^ error: several effects have an operation of this name; write the effect too, like `State.get()`
  effect State.get(), k => resume(k, 0),
  effect Config.get(key), k => resume(k, key),
};

================================================================================
A handler needs arms for all the operations of its effects
:test-prelude
================================================================================
effect State { fn get() -> i32; fn set(value: i32); fn reset(); }
effect Log { fn info(m: String); fn warn(m: String); }
fn run(body: fn() -> i32 does State + Log) -> i32 = match body() {
  v => v,
  effect get(), k => resume(k, 0),
       //^^^^^ error: this match handles `State`, so it needs arms for `set` and `reset` too
  effect info(m), k => resume(k, ()),
       //^^^^^^^ error: this match handles `Log`, so it needs an arm for `warn` too
};

================================================================================
The arms of each operation cover its arguments
:test-prelude
================================================================================
effect Yield { fn yield(value: i32); }
effect Ask { fn ask(question: Option[i32], quiet: Bool) -> i32; }
fn run(body: fn() -> i32 does Yield + Ask) -> i32 = match body() {
  v => v,
  effect yield(0), k => resume(k, ()),
       //^^^^^^^^ warning: the arms for `yield` don't cover every value: `yield(_)` matches none of them
  effect ask(Some(_), _) | ask(None, True), k => resume(k, 0),
       //^^^^^^^^^^^^^^^ warning: the arms for `ask` don't cover every value: `ask(Option.None, False)` matches none of them
  effect ask(Some(1), q), k => resume(k, 1),
       //^^^^^^^^^^^^^^^ warning: this arm is never taken: the arms before it match every value it does
};

================================================================================
A guard doesn't cover an operation's arguments
:test-prelude
================================================================================
effect Log { fn log(m: i32); }
fn run(body: fn() -> i32 does Log, quiet: Bool) -> i32 = match body() {
  v => v,
  effect log(m), k if quiet => resume(k, ()),
       //^^^^^^ warning: the arms for `log` don't cover every value: `log(_)` matches none of them
};
