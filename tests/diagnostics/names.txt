================================================================================
Unresolved name does not cascade
:test-prelude
================================================================================
  foo + 1;
//^^^ error: cannot find `foo`
  True == foo;
        //^^^ error: cannot find `foo`

================================================================================
Unresolved type is reported without its arguments
================================================================================
  let x: List[T, i32] = 1;
       //^^^^ error: cannot find type `List`
            //^ error: cannot find type `T`

================================================================================
Duplicate definitions
================================================================================
fn f() {}
const f: i32 = 1;
    //^ error: `f` is defined more than once

================================================================================
The unknown part of a path is reported
================================================================================
fn push(x: i32, y: i32) = ();
fn f(b: i32) { self.chunks.push(1); }
             //^^^^ error: cannot find `self`

================================================================================
Unknown fields are reported at their name
================================================================================
  struct Point { x: i32, y: i32 }
  let z = 3;
  Point { x: 1, y: 2, w: 3 };
                    //^ error: there is no field `w`
  Point { x: 1, y: 2, z };
                    //^ error: there is no field `z`
  fn f(p: Point) = match p { Point { x, v, .. } => x };
                                      //^ error: there is no field `v`

================================================================================
The alternatives of an or-pattern bind the same names
:test-prelude
================================================================================
enum E { A(i32), B(i32), C(Bool) }
fn f(e: E) -> i32 = match e { E.A(x) | E.B(x) => x, E.C(_) => 0 };
fn g(e: E) -> i32 = match e { E.A(x) | E.C(_) => 1, E.B(_) => 2 };
                                //^ error: `x` is bound in some alternatives of this pattern, but not in all of them
fn h(e: E) -> i32 = match e { E.A(_) | E.B(y) => 1, E.C(_) => 2 };
                                         //^ error: `y` is bound in some alternatives of this pattern, but not in all of them
fn i(e: E) -> i32 = match e { E.A(mut x) | E.B(x) => x, E.C(_) => 0 };
                                             //^ error: `x` is `mut` in some alternatives of this pattern, but not in all of them

================================================================================
A name has one type in all the alternatives
:test-prelude
================================================================================
enum E { A(i32), B(Bool), C }
fn f(e: E) = match e { E.A(x) | E.B(x) => 1, E.C => 0 };
                                  //^ error: expected `i32`, found `Bool`

================================================================================
The operations of an effect arm bind the same names
:test-prelude
================================================================================
effect Log { fn info(m: String); fn warn(m: String); fn note(m: String); fn tick(); }
fn f(body: fn() does Log) -> () does Log = match body() {
  v => v,
  effect info(m) | warn(m), _ => (),
  effect tick() | note(n), _ => (),
                     //^ error: `n` is bound in some alternatives of this pattern, but not in all of them
};
