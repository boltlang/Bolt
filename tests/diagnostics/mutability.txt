Changing structs that are not mut
================================================================================
struct Point { x: i32, y: i32 }
let p = Point { x: 1, y: 2 };
  p.x = 3;
//^ error: a field of an `imm Point` is changed here, but only a `mut` struct can be changed
let q = mut Point { x: 1, y: 2 };
q.x = 3;
fn move_x(r) { r.x = 1; }
  move_x(p);
//^^^^^^ error: a field of an `imm Point` is changed here, but only a `mut` struct can be changed
struct Bag { items: mut [i32] }
                  //^^^^^^^^^ error: a field can't be `mut`: it has the mutability of its struct, so make the struct `mut` instead
let a = [mut [1]];
      //^^^^^^^^^ error: an `imm [mut [i32]]` can't hold a `mut [i32]`: what an `imm` value holds can't be changed either
fn f(x: mut i32) {}
      //^^^^^^^ error: only an array, a struct or a function can be `mut` or `imm`

================================================================================
Closure mutability
================================================================================
struct Holder { poke: fn() }
                    //^^^^ label: expected because of the type of field `poke`
struct Point { x: i32 }
let p = mut Point { x: 0 };
let h = Holder { poke: || { p.x = 1; } };
                     //^^^^^^^^^^^^^^^ error: expected `fn() -> ()`, found `mut fn() -> ()`
let fs = [|| { p.x = 1; }];
       //^^^^^^^^^^^^^^^^^ error: an `imm [mut fn() -> ()]` can't hold a `mut fn() -> ()`: what an `imm` value holds can't be changed either
struct Getter[T] { get: fn() -> T }
                      //^^^^^^^^^ label: expected because of the type of field `get`
fn copy[T](g: 'm Getter[T]) = Getter { get: g.get };
                                          //^^^^^ error: expected `fn() -> T`, found `'m fn() -> T`
