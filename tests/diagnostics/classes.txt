================================================================================
Impls must define the methods of their class
================================================================================
class Eq[T] {
  fn eq(a: T, b: T) -> Bool;
  fn ne(a: T, b: T) -> Bool = eq(b, a);
}
class Ord[T] { fn lt(a: T, b: T) -> Bool; fn gt(a: T, b: T) -> Bool; }
impl Eq[i32] { fn eq(a, b) = True; }
impl Eq[Bool] {}
   //^^^^^^^^ error: missing method `eq` of class `Eq`
impl Ord[i32];
   //^^^^^^^^ error: missing methods `lt`, `gt` of class `Ord`

            //^^^^^ label: expected because of the declaration in class `Sized`

================================================================================
Overlapping impls
================================================================================
class Size[T] { fn size(x: T) -> i32; }
impl[T] Size[[T]] { fn size(x) = 1; }
      //^^^^^^^^^ label: the impl it overlaps
impl Size[[i32]] { fn size(x) = 2; }
   //^^^^^^^^^^^ error: the impl for `Size[[i32]]` overlaps the one for `Size[[T]]`: some types fit both, so a constraint can't choose one; remove one of them, or make their instances differ
  size([1]);
//^^^^ error: more than one impl fits `Size[imm [i32]]`, so it's not known which to use; their instances overlap

================================================================================
Unused impl params
================================================================================
class Show[T] { fn show(x: T) -> i32; }
struct Box[T] { item: T }
impl[T, U] Show[Box[T]] { fn show(x) = 1; }
      //^ error: `U` isn't in the instance `Show[Box[T]]`, so choosing the impl can't decide it; use it in the instance, or remove it
impl[T, 'v] Show[[T]] { fn show(x) = 2; }
      //^^ error: `'v` isn't in the instance `Show[[T]]`, so choosing the impl can't decide it; write it in the instance, like `'v [T]`, or remove it
impl Show[i32] where 'q: imm { fn show(x) = 3; }
                   //^^ error: `'q` isn't in the instance `Show[i32]`, so choosing the impl can't decide it; write it before an array or struct in the instance, or remove it
