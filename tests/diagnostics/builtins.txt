:prelude

//- /toolchain/std/__prelude__.bolt
pub imm struct String(i32);
//- /main.bolt
// A builtin's string is the prelude's `String`, so this has a prelude
// that has one.
extern "bolt" {
  fn print(text: String);
  fn prnt(text: String);
   //^^^^ error: there is no builtin `prnt`; the builtins are `print`, `panic`, `array_len`, `array_alloc`, `array_get`, `array_set`, `array_copy`, `array_freeze`, and the operations of the primitive types, like `i32_add`
  fn panic(message: String) -> i32;
   //^^^^^ error: the builtin `panic` must have the type `fn(String) -> !`
}

The type parameters of a builtin can have any name
================================================================================
extern "bolt" { fn array_len[Elem](items: [Elem]) -> i32; }

================================================================================
extern "bolt" {
  fn array_len[T](xs: mut [T]) -> i32;
   //^^^^^^^^^ error: the builtin `array_len` must have the type `fn([a]) -> i32`
}

  fn array_len[T](xs: [T]) -> T;
