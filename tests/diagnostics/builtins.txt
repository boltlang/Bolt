================================================================================
Builtin declarations
:prelude
================================================================================
//- /toolchain/std/__prelude__.bolt
pub imm struct String(i32);
//- /main.bolt
// A builtin's string is the prelude's `String`, so this has a prelude
// that has one.
#[builtin]
fn print(text: String);
#[builtin]
fn prnt(text: String);
 //^^^^ error: there is no builtin `prnt`; the builtins are `print`, `abort`, `array_len`, `array_alloc`, `array_get`, `array_set`, `array_copy`, `array_freeze`, `resume`, and the operations of the primitive types, like `i32_add`
#[builtin]
fn abort(code: i32) -> !;
 //^^^^^ error: the builtin `abort` must have the type `fn() -> !`

================================================================================
The type parameters of a builtin can have any name
================================================================================
#[builtin] fn array_len[Elem](items: [Elem]) -> i32;

================================================================================
A builtin array_len that takes a mut array
================================================================================
#[builtin]
fn array_len[T](xs: mut [T]) -> i32;
 //^^^^^^^^^ error: the builtin `array_len` must have the type `fn([a]) -> i32`

================================================================================
A builtin array_len that gives an element
================================================================================
#[builtin]
fn array_len[T](xs: [T]) -> T;
 //^^^^^^^^^ error: the builtin `array_len` must have the type `fn([a]) -> i32`

================================================================================
A builtin array_alloc that gives a mut array
================================================================================
#[builtin] fn array_alloc[T](len: i32) -> mut [T];

================================================================================
A builtin array_alloc that gives an imm array
================================================================================
// Without `mut`, the result would be `imm`, and the elements could never
// be set.
#[builtin]
fn array_alloc[T](len: i32) -> [T];
 //^^^^^^^^^^^ error: the builtin `array_alloc` must have the type `fn(i32) -> mut [a]`

//- /main.bolt

================================================================================
A function without a body has to be a builtin
================================================================================
fn nothing(x: i32);
 //^^^^^^^ error: this function has no body; only a function the compiler provides, with `#[builtin]`, or the host, with `#[link(..)]`, can do without one

================================================================================
A builtin can't have a body
================================================================================
#[builtin]
fn i32_add(a: i32, b: i32) -> i32 { a }
 //^^^^^^^ error: a builtin can't have a body: the compiler provides it

================================================================================
A builtin has to be at the top level
================================================================================
struct Num(i32);
#[builtin]
fn Num.i32_add(a: i32, b: i32) -> i32;
     //^^^^^^^ error: a builtin must be a function at the top level of a file, not in a class, an impl, an `extern` block or the namespace of a type

================================================================================
A function linked to the host
================================================================================
#[link(wasm, module = "env")]
fn log(x: i32);
#[link(wasm, module = "wasi_snapshot_preview1", name = "proc_exit")]
fn exit(code: i32) -> !;

================================================================================
A function linked to the host can't have a body or be a builtin
================================================================================
#[link(wasm, module = "env")]
fn log(x: i32) {}
 //^^^ error: a function linked to the host can't have a body: the host provides it
#[builtin]
#[link(wasm, module = "env")]
fn print(x: i32);
 //^^^^^ error: a function can't be both a builtin and linked to the host

================================================================================
How link is written
================================================================================
#[link]
fn a(x: i32);
 //^ error: `#[link]` must say what the function is linked to first; write it like `#[link(wasm, module = "env", name = "log")]`
#[link(native, module = "c")]
fn b(x: i32);
 //^ error: a function can't be linked to `native`: only `wasm` imports are supported; write it like `#[link(wasm, module = "env", name = "log")]`
#[link(wasm)]
fn c(x: i32);
 //^ error: `#[link(wasm, ..)]` needs the Wasm module of the import; write it like `#[link(wasm, module = "env", name = "log")]`
#[link(wasm, module = "env", lib = "m")]
fn d(x: i32);
 //^ error: `#[link]` has no `lib`; it takes `module` and `name`
#[link(wasm, module = "env", module = "env")]
fn e(x: i32);
 //^ error: `module` is given twice
#[link(wasm, module = env)]
fn f(x: i32);
 //^ error: after `wasm`, `#[link]` takes `key = "string"` pairs; write it like `#[link(wasm, module = "env", name = "log")]`
