================================================================================
A function without a body has to be a builtin
================================================================================
fn nothing(x: i32);
 //^^^^^^^ error: this function has no body; only a function the compiler provides, with `#[builtin]`, or the host, with `#[link(..)]`, can do without one

================================================================================
A builtin can't have a body
================================================================================
#[builtin]
fn i32_add(a: i32, b: i32) -> i32 { a }
 //^^^^^^^ error: a builtin can't have a body: the compiler provides it

================================================================================
A builtin has to be at the top level
================================================================================

================================================================================
A function linked to the host can't have a body or be a builtin
================================================================================
#[link(wasm, module = "env")]
fn log(x: i32) {}
 //^^^ error: a function linked to the host can't have a body: the host provides it
#[builtin]
#[link(wasm, module = "env")]
fn print(x: i32);
 //^^^^^ error: a function can't be both a builtin and linked to the host

================================================================================
How link is written
================================================================================
#[link]
fn a(x: i32);
 //^ error: `#[link]` must say what the function is linked to first; write it like `#[link(wasm, module = "env", name = "log")]`
#[link(native, module = "c")]
fn b(x: i32);
 //^ error: a function can't be linked to `native`: it's linked to `wasm`, a Wasm import, or `emu`, the evaluator; write it like `#[link(wasm, module = "env", name = "log")]`
#[link(wasm)]
fn c(x: i32);
 //^ error: `#[link(wasm, ..)]` needs the Wasm module of the import; write it like `#[link(wasm, module = "env", name = "log")]`
#[link(wasm, module = "env", lib = "m")]
fn d(x: i32);
 //^ error: `#[link]` has no `lib`; it takes `module`, `name` and `abi`
#[link(wasm, module = "env", module = "env")]
fn e(x: i32);
 //^ error: `module` is given twice
#[link(wasm, module = env)]
fn f(x: i32);
 //^ error: after `wasm`, `#[link]` takes `key = "string"` pairs, and `abi = flat`; write it like `#[link(wasm, module = "env", name = "log")]`
#[link(emu, module = "env")]
fn g(x: i32);
 //^ error: `#[link(emu, ..)]` has no module: the evaluator provides it by its `name` alone
#[link(c, symbol = "puts")]
fn h(x: i32);
 //^ error: a function can't be linked to `c` yet: that comes with the native target
#[link(emu)]
fn i(x: i32);
#[link(emu, name = "j2", abi = flat)]
fn j(x: i32);
