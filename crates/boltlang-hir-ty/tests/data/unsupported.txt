================================================================================
Unsupported expressions have the error type
================================================================================
fn f(x) { x?; }
fn g(x) { x.await; }
fn h() = 1..2;
fn k() { yield 1; }

--------------------------------------------------------------------------------
f: forall a. fn(a) -> ()
g: forall a. fn(a) -> ()
h: forall a. fn() -> a
k: fn() -> ()
--------------------------------------------------------------------------------
warning at `x?`: unsupported
warning at `x.await`: unsupported
warning at `1..2`: unsupported
warning at `yield 1`: unsupported

================================================================================
Unsupported types have the error type
================================================================================
fn f(x: impl Show) = x;
fn g[T](x: T.Item) = x;

--------------------------------------------------------------------------------
f: forall a. fn({error}) -> a
g: forall T a. fn({error}) -> a
--------------------------------------------------------------------------------
warning at `impl Show`: unsupported
warning at `T.Item`: unsupported

================================================================================
Arithmetic on any type is accepted until there are numeric classes
================================================================================
"a" + "b";
true * false;
-"a";
let x = 1u8;
-x;

--------------------------------------------------------------------------------
"a" + "b": str
true * false: bool
-"a": str
let x: u8
-x: u8
