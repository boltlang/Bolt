================================================================================
Functions used before their definition
================================================================================
fn pair(x) = (first(x), first(true));
fn first(y) = y;
--------------------------------------------------------------------------------
pair: forall a. fn(a) -> (a, bool)
first: forall a. fn(a) -> a

================================================================================
Mutual recursion
================================================================================
fn even(n) = if n == 0 { true } else { odd(n - 1) };
fn odd(n) = if n == 0 { false } else { even(n - 1) };
--------------------------------------------------------------------------------
even: fn(i32) -> bool
odd: fn(i32) -> bool

================================================================================
Method call syntax
================================================================================
struct Counter { n: i32 }
fn get(c: Counter) -> i32 = c.n;
fn inc(x: i32) -> i32 = x + 1;
let c = Counter { n: 1 };
c.get().inc();
c.n.inc();
get(c);
--------------------------------------------------------------------------------

inc: fn(i32) -> i32
let c: Counter
