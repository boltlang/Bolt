================================================================================
Generalizes functions without signatures
================================================================================
fn id(x) { return x }
id(1) + 1;
id("foo") + 1;
let b = id(true);
--------------------------------------------------------------------------------
id: forall a. fn(a) -> a
id(1) + 1: i32
id("foo") + 1: str
let b: bool
--------------------------------------------------------------------------------
error at `1`: expected str, found i32

================================================================================
Functions used before their definition
================================================================================
fn pair(x) = (first(x), first(true));
fn first(y) = y;
--------------------------------------------------------------------------------
pair: forall a. fn(a) -> (a, bool)
first: forall a. fn(a) -> a

================================================================================
Mutual recursion
================================================================================
fn even(n) = if n == 0 { true } else { odd(n - 1) };
fn odd(n) = if n == 0 { false } else { even(n - 1) };
--------------------------------------------------------------------------------
even: fn(i32) -> bool
odd: fn(i32) -> bool

================================================================================
Declared signatures are checked
================================================================================
fn f(x: i32) -> bool { x }
fn g[T](x: T) -> T = x;
g(1u8);
f(true);
--------------------------------------------------------------------------------
f: fn(i32) -> bool
g: forall T. fn(T) -> T
g(1u8): u8
f(true): bool
--------------------------------------------------------------------------------
error at `x`: expected bool, found i32
error at `true`: expected i32, found bool

================================================================================
What is only unknown because of an error becomes the error type
================================================================================
fn f() = nope;
fn g(c: bool) = if c { nope } else { 1 };
fn h(x) = { nope(x); x };
--------------------------------------------------------------------------------
f: fn() -> {error}
g: fn(bool) -> i32
h: forall a. fn(a) -> a
--------------------------------------------------------------------------------
error at `nope`: unresolved
error at `nope`: unresolved
error at `nope`: unresolved
