================================================================================
Class constraints are inferred and solved
================================================================================
class Show[T] { fn show(x: T) -> str; }
impl Show[i32] { fn show(x) = "int"; }
fn twice(x) = (show(x), show(x));
twice(1);
twice(true);
--------------------------------------------------------------------------------
Show[i32].show: fn(i32) -> str
twice: forall a. Show[a] => fn(a) -> (str, str)
twice(1): (str, str)
twice(true): (str, str)
--------------------------------------------------------------------------------
error at `twice`: no instance Show[bool]

================================================================================
Where clauses are assumed
================================================================================
class Eq[T] { fn eq(a: T, b: T) -> bool; }
class Ord[T] where Eq[T] { fn lt(a: T, b: T) -> bool; }
fn le[T](a: T, b: T) -> bool where Ord[T] = lt(a, b) || eq(a, b);
fn bad[T](a: T) -> bool = eq(a, a);
--------------------------------------------------------------------------------
le: forall T. Ord[T] => fn(T, T) -> bool
bad: forall T. fn(T) -> bool
--------------------------------------------------------------------------------
error at `eq`: no instance Eq[T]

================================================================================
Classes with several parameters
================================================================================
class Convert[A, B] { fn convert(a: A) -> B; }
impl Convert[i32, str] { fn convert(a) = "int"; }
fn to_str(x: i32) -> str = convert(x);
fn to_bool(x: i32) -> bool = convert(x);

--------------------------------------------------------------------------------
Convert[i32, str].convert: fn(i32) -> str
to_str: fn(i32) -> str
to_bool: fn(i32) -> bool
--------------------------------------------------------------------------------
error at `convert`: no instance Convert[i32, bool]

================================================================================
Generic impls
================================================================================
class Show[T] { fn show(x: T) -> str; }
enum Option[T] { Some(T), None }
impl Show[i32] { fn show(x) = "int"; }
impl[T] Show[Option[T]] where Show[T] { fn show(x) = "option"; }
show(Option.Some(1));
show(Option.Some(true));

--------------------------------------------------------------------------------
Show[i32].show: fn(i32) -> str
Show[Option[T]].show: forall T. Show[T] => fn(Option[T]) -> str
show(Option.Some(1)): str
show(Option.Some(true)): str
--------------------------------------------------------------------------------
error at `show`: no instance Show[bool]

================================================================================
Superclasses are assumed
================================================================================
class Eq[T] { fn eq(a: T, b: T) -> bool; }
class Ord[T] where Eq[T] { fn lt(a: T, b: T) -> bool; }
fn max[T](a: T, b: T) -> T where Ord[T] = if lt(a, b) || eq(a, b) { b } else { a };

--------------------------------------------------------------------------------
max: forall T. Ord[T] => fn(T, T) -> T

================================================================================
Constraints are inferred from several methods
================================================================================
class Show[T] { fn show(x: T) -> str; }
class Eq[T] { fn eq(a: T, b: T) -> bool; }
fn show_if_same(a, b) = if eq(a, b) { show(a) } else { "different" };

--------------------------------------------------------------------------------
show_if_same: forall a. Eq[a] => Show[a] => fn(a, a) -> str

================================================================================
A constraint on a type that is never known is ambiguous
================================================================================
class Default[T] { fn default() -> T; }
impl Default[i32] { fn default() = 0; }
fn f() { default(); }
fn g() -> i32 = default();

--------------------------------------------------------------------------------
Default[i32].default: fn() -> i32
f: fn() -> ()
g: fn() -> i32
--------------------------------------------------------------------------------
error at `default`: ambiguous
