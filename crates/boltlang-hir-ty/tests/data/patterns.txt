================================================================================
Tuple patterns
================================================================================
let (a, b) = (1, true);
let (c, .., d) = (1, 2, "x");
let (e, f) = (1, 2, 3);

--------------------------------------------------------------------------------
let (a, b): (i32, bool)
let (c, .., d): (i32, i32, str)
let (e, f): (i32, i32, i32)
--------------------------------------------------------------------------------
error at `(e, f)`: expected (i32, i32, i32), found (a, b)

================================================================================
Slice patterns
================================================================================
let [a, b] = [1, 2];
let [first, .., last] = ["a", "b", "c"];
let [x] = (1,);

--------------------------------------------------------------------------------
let [a, b]: [i32]
let [first, .., last]: [str]
let [x]: (i32,)
--------------------------------------------------------------------------------
error at `[x]`: expected (i32,), found [a]

================================================================================
Wildcards and bindings with @
================================================================================
let _ = 1;
let whole @ (x, _) = (1, 2);

--------------------------------------------------------------------------------
let _: i32
let whole @ (x, _): (i32, i32)

================================================================================
Literal patterns
================================================================================
fn f(x) = match x { 0 => "zero", -1 => "minus one", _ => "other" };
fn g(s) = match s { "a" => 1, _ => 2 };
fn h(x: i32) = match x { "a" => 1, _ => 2 };

--------------------------------------------------------------------------------
f: fn(i32) -> str
g: fn(str) -> i32
h: fn(i32) -> i32
--------------------------------------------------------------------------------
error at `"a"`: expected i32, found str

================================================================================
Range patterns
================================================================================
fn f(x) = match x { 0..=9 => "digit", 10.. => "more", _ => "negative" };
fn g(c: char) = match c { 'a'..='z' => true, _ => false };

--------------------------------------------------------------------------------
f: fn(i32) -> str
g: fn(char) -> bool

================================================================================
Or patterns
================================================================================
fn f(x) = match x { 1 | 2 | 3 => "small", _ => "big" };
fn g(x: i32) = match x { 1 | "two" => 1, _ => 2 };

--------------------------------------------------------------------------------
f: fn(i32) -> str
g: fn(i32) -> i32
--------------------------------------------------------------------------------
error at `"two"`: expected i32, found str

================================================================================
Variant patterns
================================================================================
enum Shape { Circle(f64), Rect(f64, f64), Empty }
fn area(s) = match s {
  Shape.Circle(r) => r * r,
  Shape.Rect(w, h) => w * h,
  Shape.Empty => 0.0,
};

--------------------------------------------------------------------------------
area: fn(Shape) -> f64

Record patterns
================================================================================

fn x_of(p) = match p { Point { x, .. } => x };

================================================================================
Patterns in parameters
================================================================================
fn first((a, _)) = a;
fn swap((a, b): (i32, bool)) = (b, a);

--------------------------------------------------------------------------------
first: forall a b. fn((a, b)) -> a
swap: fn((i32, bool)) -> (bool, i32)

================================================================================
If let and let else
================================================================================
enum Option[T] { Some(T), None }
fn get(o, default) = if let Option.Some(x) = o { x } else { default };
fn unwrap(o: Option[i32]) -> i32 {
  let Option.Some(x) = o else { return 0; };
  x
}

--------------------------------------------------------------------------------
get: forall a. fn(Option[a], a) -> a
unwrap: fn(Option[i32]) -> i32

================================================================================
A pattern that can't match the type
================================================================================

fn f(p: Point) = match p { (a, b) => a };
fn g(x: i32) = match x { Point { x, .. } => x };

--------------------------------------------------------------------------------
f: fn(Point) -> {error}
g: fn(i32) -> i32
--------------------------------------------------------------------------------
error at `(a, b)`: expected Point, found (a, b)
error at `Point { x, .. }`: expected i32, found Point
