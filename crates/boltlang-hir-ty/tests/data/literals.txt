================================================================================
Integer literals default to i32
================================================================================
let a = 1;
let b = 0xff;
1 + 2;

--------------------------------------------------------------------------------
let a: i32
let b: i32
1 + 2: i32

================================================================================
Float literals default to f64
================================================================================
let a = 1.5;
1.0 * 2.0;

--------------------------------------------------------------------------------
let a: f64
1.0 * 2.0: f64

================================================================================
Suffixes give the type
================================================================================
let a = 1u8;
let b = 2i64;
let c = 1.5f32;

--------------------------------------------------------------------------------
let a: u8
let b: i64
let c: f32

================================================================================
An unsuffixed literal takes the type it's used as
================================================================================
fn f(x: u64) -> u64 = x;
f(1);
let a: u16 = 1;
let b = 1u8 + 2;
let c: f32 = 1.5;

--------------------------------------------------------------------------------
f: fn(u64) -> u64
f(1): u64
let a: u16
let b: u8
let c: f32

================================================================================
An integer literal is not a float
================================================================================
let a: f64 = 1;
let b: i32 = 1.5;
1 + 1.5;

--------------------------------------------------------------------------------
let a: f64
let b: i32
1 + 1.5: i32
--------------------------------------------------------------------------------
error at `1`: expected f64, found i32
error at `1.5`: expected i32, found f64
error at `1.5`: expected i32, found f64

================================================================================
Other literals
================================================================================
let a = true;
let b = 'x';
let c = b'x';
let d = "text";
let e = b"bytes";

--------------------------------------------------------------------------------
let a: bool
let b: char
let c: u8
let d: str
let e: [u8]

================================================================================
Literals that don't fit their type
================================================================================
let a: bool = 1;
let b: str = 'x';
let c: char = "x";

--------------------------------------------------------------------------------
let a: bool
let b: str
let c: char
--------------------------------------------------------------------------------
error at `1`: expected bool, found i32
error at `'x'`: expected str, found char
error at `"x"`: expected char, found str
